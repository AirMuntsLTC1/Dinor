class AuthGate extends StatelessWidget {

    final auth = Supabase.instance.client.auth;
    return StreamBuilder<AuthState>(
      stream: auth.onAuthStateChange,
      // Seed with the current session so the right screen shows on the first frame.
      initialData: AuthState(AuthChangeEvent.initialSession, auth.currentSession),
      builder: (context, snapshot) {

import 'dart:convert';
import 'dart:ffi';

import 'package:ffi/ffi.dart';

// ---- C signatures ----------------------------------------------------------
typedef _VersionC = Pointer<Utf8> Function();
typedef _VersionD = Pointer<Utf8> Function();

typedef _Utf8LenC = Int32 Function(Pointer<Utf8>);
typedef _Utf8LenD = int Function(Pointer<Utf8>);

typedef _TextFnC = Int32 Function(Pointer<Utf8>, Pointer<Utf8>, Int32);
typedef _TextFnD = int Function(Pointer<Utf8>, Pointer<Utf8>, int);

typedef _SearchC = Int32 Function(Pointer<Utf8>, Pointer<Utf8>);
typedef _SearchD = int Function(Pointer<Utf8>, Pointer<Utf8>);

typedef _ExtractC = Int32 Function(Pointer<Utf8>, Pointer<Utf8>, Pointer<Utf8>, Int32);
typedef _ExtractD = int Function(Pointer<Utf8>, Pointer<Utf8>, Pointer<Utf8>, int);

typedef _HashC = Int32 Function(Pointer<Utf8>);
typedef _HashD = int Function(Pointer<Utf8>);

typedef _AnimationC = Float Function(Double, Int32);
typedef _AnimationD = double Function(double, int);

typedef _StatusC = Pointer<Utf8> Function(Int32);
typedef _StatusD = Pointer<Utf8> Function(int);

typedef _TimestampC = Int64 Function();
typedef _TimestampD = int Function();

typedef _StaleC = Int32 Function(Int64, Int64, Int64);
typedef _StaleD = int Function(int, int, int);

typedef _RetryC = Int64 Function(Int32, Int64, Int64);
typedef _RetryD = int Function(int, int, int);

/// Thin, cached wrapper around libdinor_core.so.
///
/// * One library handle for the whole app (never reopened per frame).
/// * Buffers are sized from UTF-8 byte length, not Dart's UTF-16 length.
/// * If the native library is missing, every method falls back to pure Dart
///   so the app keeps working instead of crashing.
class DinorCore {
  DinorCore._(DynamicLibrary? lib)
      : _lib = lib,
        _version = lib == null ? null : _try(() => lib.lookupFunction<_VersionC, _VersionD>('dinor_version')),
        _utf8Len = lib == null ? null : _try(() => lib.lookupFunction<_Utf8LenC, _Utf8LenD>('dinor_utf8_length')),
        _encrypt = lib == null ? null : _try(() => lib.lookupFunction<_TextFnC, _TextFnD>('dinor_encrypt')),
        _format = lib == null ? null : _try(() => lib.lookupFunction<_TextFnC, _TextFnD>('dinor_format_message')),
        _sanitize = lib == null ? null : _try(() => lib.lookupFunction<_TextFnC, _TextFnD>('dinor_sanitize_input')),
        _batchSanitize = lib == null ? null : _try(() => lib.lookupFunction<_TextFnC, _TextFnD>('dinor_batch_sanitize')),
        _escapeLike = lib == null ? null : _try(() => lib.lookupFunction<_TextFnC, _TextFnD>('dinor_escape_like')),
        _search = lib == null ? null : _try(() => lib.lookupFunction<_SearchC, _SearchD>('dinor_fast_search')),
        _extract = lib == null ? null : _try(() => lib.lookupFunction<_ExtractC, _ExtractD>('dinor_extract_web_answer')),
        _hash = lib == null ? null : _try(() => lib.lookupFunction<_HashC, _HashD>('dinor_calculate_hash')),
        _animation = lib == null ? null : _try(() => lib.lookupFunction<_AnimationC, _AnimationD>('dinor_animation_scale')),
        _status = lib == null ? null : _try(() => lib.lookupFunction<_StatusC, _StatusD>('dinor_get_status_icon')),
        _timestamp = lib == null ? null : _try(() => lib.lookupFunction<_TimestampC, _TimestampD>('dinor_fast_timestamp')),
        _stale = lib == null ? null : _try(() => lib.lookupFunction<_StaleC, _StaleD>('dinor_is_stale')),
        _retry = lib == null ? null : _try(() => lib.lookupFunction<_RetryC, _RetryD>('dinor_retry_delay_ms'));

  static DinorCore? _instance;

  /// Looks up one symbol; a missing symbol yields null (Dart fallback) instead of throwing.
  static T? _try<T>(T Function() lookup) {
    try {
      return lookup();
    } catch (_) {
      return null;
    }
  }

  final DynamicLibrary? _lib;

  final _VersionD? _version;
  final _Utf8LenD? _utf8Len;
  final _TextFnD? _encrypt;
  final _TextFnD? _format;
  final _TextFnD? _sanitize;
  final _TextFnD? _batchSanitize;
  final _TextFnD? _escapeLike;
  final _SearchD? _search;
  final _ExtractD? _extract;
  final _HashD? _hash;
  final _AnimationD? _animation;
  final _StatusD? _status;
  final _TimestampD? _timestamp;
  final _StaleD? _stale;
  final _RetryD? _retry;

  /// True when libdinor_core.so loaded successfully.
  bool get isNative => _lib != null;

  /// Returns the shared instance. Safe to call on every frame: it is cached,
  /// and the library is opened at most once per app run.
  factory DinorCore.load() {
    final existing = _instance;
    if (existing != null) return existing;
    DynamicLibrary? lib;
    try {
      lib = DynamicLibrary.open('libdinor_core.so');
    } catch (_) {
      lib = null; // no native library: use Dart implementations
    }
    return _instance = DinorCore._(lib);
  }

  // ---- helpers -------------------------------------------------------------
  static int _bytes(String s) => utf8.encode(s).length;

  String _runText(_TextFnD? fn, String input, {int? outFactor}) {
    if (fn == null) return input;
    final inPtr = input.toNativeUtf8();
    final outSize = _bytes(input) * (outFactor ?? 1) + 1;
    final out = calloc<Uint8>(outSize).cast<Utf8>();
    try {
      final n = fn(inPtr, out, outSize);
      return n < 0 ? input : out.toDartString();
    } finally {
      calloc.free(inPtr);
      calloc.free(out);
    }
  }

  // ---- identity ------------------------------------------------------------
  String version() {
    if (_version == null) return 'Dinor (Dart fallback)';
    return _version().toDartString();
  }

  int utf8Length(String text) {
    if (_utf8Len == null) return _bytes(text);
    final p = text.toNativeUtf8();
    try {
      return _utf8Len(p);
    } finally {
      calloc.free(p);
    }
  }

  // ---- text ----------------------------------------------------------------
  /// Reversible obfuscation (not encryption) kept for compatibility.
  String encrypt(String text) {
    if (_encrypt == null) return String.fromCharCodes(text.runes.toList().reversed);
    return _runText(_encrypt, text);
  }

  bool search(String haystack, String needle) {
    if (_search == null) {
      return haystack.toLowerCase().contains(needle.toLowerCase());
    }
    final h = haystack.toNativeUtf8();
    final n = needle.toNativeUtf8();
    try {
      return _search(h, n) == 1;
    } finally {
      calloc.free(h);
      calloc.free(n);
    }
  }

  String extractAiAnswer(String query, String webText) {
    if (_extract == null) return '';
    final q = query.toNativeUtf8();
    final w = webText.toNativeUtf8();
    const outSize = 512;
    final out = calloc<Uint8>(outSize).cast<Utf8>();
    try {
      final n = _extract(q, w, out, outSize);
      return n < 0 ? '' : out.toDartString();
    } finally {
      calloc.free(q);
      calloc.free(w);
      calloc.free(out);
    }
  }

  String sanitizeInput(String text) {
    if (_sanitize == null) {
      return text.replaceAll(RegExp(r'[\u0000-\u0009\u000B-\u001F\u007F]'), '');
    }
    return _runText(_sanitize, text);
  }

  /// Sanitizes, collapses whitespace and trims. Used before every send.
  String batchSanitize(String text) {
    if (_batchSanitize == null) {
      return text.replaceAll(RegExp(r'[\r\n\t]+'), ' ').replaceAll(RegExp(r' {2,}'), ' ').trim();
    }
    return _runText(_batchSanitize, text);
  }

  String escapeLike(String text) {
    if (_escapeLike == null) {
      return text.replaceAllMapped(RegExp(r'[%_\\]'), (m) => '\\${m[0]}');
    }
    return _runText(_escapeLike, text, outFactor: 2);
  }

  String formatMessage(String text) => _runText(_format, text);

  int calculateHash(String text) {
    if (_hash == null) {
      var h = 5381;
      for (final c in utf8.encode(text)) {
        h = ((h << 5) + h + c) & 0x7FFFFFFF;
      }
      return h;
    }
    final p = text.toNativeUtf8();
    try {
      return _hash(p);
    } finally {
      calloc.free(p);
    }
  }

  // ---- timing / cache ------------------------------------------------------
  int getFastTimestamp() {
    if (_timestamp == null) return DateTime.now().millisecondsSinceEpoch;
    return _timestamp();
  }

  /// True when a cache entry saved at [savedAtMs] is older than [ttlMs].
  bool isStale({required int savedAtMs, required int ttlMs, int? nowMs}) {
    final now = nowMs ?? DateTime.now().millisecondsSinceEpoch;
    if (_stale == null) {
      if (savedAtMs <= 0 || ttlMs <= 0 || now < savedAtMs) return true;
      return now - savedAtMs > ttlMs;
    }
    return _stale(savedAtMs, ttlMs, now) == 1;
  }

  /// Exponential backoff delay in ms for retry [attempt] (0-based).
  int retryDelayMs(int attempt, {int baseMs = 250, int capMs = 4000}) {
    if (_retry == null) {
      final shift = attempt < 0 ? 0 : (attempt > 30 ? 30 : attempt);
      final d = baseMs * (1 << shift);
      return d > capMs ? capMs : d;
    }
    return _retry(attempt, baseMs, capMs);
  }

  // ---- animation / status --------------------------------------------------
  double getAnimationScale(double timeSeconds, int dotIndex) {
    if (_animation == null) {
      final delay = dotIndex * 0.2;
      var progress = (timeSeconds - delay) % 1.0;
      if (progress < 0) progress += 1.0;
      return 1.0 + (0.6 * (progress < 0.5 ? progress * 2 : (1 - progress) * 2));
    }
    return _animation(timeSeconds, dotIndex);
  }

  String getStatusIcon(int statusCode) {
    if (_status == null) {
      switch (statusCode) {
        case 0: return 'sending';
        case 1: return 'sent';
        case 2: return 'failed';
        default: return 'unknown';
      }
    }
    return _status(statusCode).toDartString();
  }
}

class _HomeScreenState extends State<HomeScreen> {
  int _currentIndex = 0;
  int _drawerPage = 0;
  final supabase = Supabase.instance.client;

  List<Map<String, dynamic>> _conversations = [];
  List<Map<String, dynamic>> _searchResults = [];
  bool _isSearching = false;
  bool _isLoading = true; // true only when there is no cached list to show
  final FocusNode _searchFocus = FocusNode();

  RealtimeChannel? _convChannel;
  Timer? _refreshDebounce;
  Timer? _searchDebounce;
  int _fetchSeq = 0;   // drops out-of-order conversation responses
  int _searchSeq = 0;  // drops out-of-order search responses
  bool _incomingDialogOpen = false;
  BuildContext? _dialogContext;
  Timer? _homePoll;

  @override
  void initState() {
    super.initState();

    // 1. Paint instantly from the local cache (no spinner on startup).
    final cached = LocalCache.peekConversations();
    if (cached != null) {
      _conversations = cached;
      _isLoading = false;
    }

    // 2. Refresh from the network in the background.
    _fetchConversations();

    // 3. Realtime updates, debounced so bursts trigger one refresh.
    _subscribeRealtime();
    // Fallback: refresh the chat list regularly, in case Realtime lags.
    _homePoll = Timer.periodic(const Duration(seconds: 6), (_) => _scheduleRefresh());

  }

  void _subscribeRealtime() {
    final me = supabase.auth.currentUser?.id;
    if (me == null) return;
    _convChannel?.unsubscribe();
    _convChannel = supabase.channel('conversations:$me').onPostgresChanges(
      event: PostgresChangeEvent.all,
      schema: 'public',
      table: 'conversations',
      callback: (_) => _scheduleRefresh(),
    ).subscribe();
  }

  void _scheduleRefresh() {
    _refreshDebounce?.cancel();
    _refreshDebounce = Timer(const Duration(milliseconds: 300), _fetchConversations);
  }

  @override
  void dispose() {
    _refreshDebounce?.cancel();
    _searchDebounce?.cancel();
    _homePoll?.cancel();
    _convChannel?.unsubscribe();
    _searchFocus.dispose();
    super.dispose();
  }

  Future<void> _fetchConversations() async {
    final seq = ++_fetchSeq;
    try {
      final data = await ChatService.getMyConversations();
      if (seq != _fetchSeq) return; // a newer fetch is already in flight
      LocalCache.saveConversations(data); // fire-and-forget persistence
      if (mounted) {
        setState(() {
          _conversations = data;
          _isLoading = false;
        });
      }
    } catch (e) {
      debugPrint('Error fetching conversations: $e');
      // Keep showing cached data on failure; only stop the skeleton.
      if (mounted) setState(() => _isLoading = false);
    }
  }

  void _onSearchChanged(String query) {
    _searchDebounce?.cancel();
    _searchDebounce = Timer(const Duration(milliseconds: 250), () => _searchUsers(query));
  }

  Future<void> _searchUsers(String query) async {
    final seq = ++_searchSeq;
    if (query.trim().isEmpty) {
      if (mounted) setState(() { _isSearching = false; _searchResults = []; });
      return;
    }
    if (mounted) setState(() => _isSearching = true);
    try {
      final results = await ChatService.searchUsers(query);
      if (seq == _searchSeq && mounted) setState(() => _searchResults = results);
    } catch (e) {
      debugPrint('Search error: $e');
      if (seq == _searchSeq && mounted) setState(() => _searchResults = []);
    }
  }

  String _initial(String? name) =>
      (name != null && name.isNotEmpty) ? name[0].toUpperCase() : '?';

  Future<void> _openChat(String otherUserId, String name, {String? knownConvId}) async {
    _searchFocus.unfocus();

    // Existing chats already know their id: open immediately, no network wait.
    String convId = knownConvId ?? '';
    if (convId.isEmpty) {
      try {
        convId = await ChatService.getOrCreateConversation(otherUserId);
      } catch (e) {
        debugPrint('Could not open chat: $e');
        if (mounted) {
          ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text("Couldn't open chat. Try again.")));
        }
        return;
      }
    }

    if (!mounted) return;
    // Clear the unread badge instantly; the list refreshes when the chat closes.
    setState(() {
      for (var conv in _conversations) {
        if (conv['id'] == convId) {
          conv['unread_count'] = 0;
          conv['latest_unread'] = null;
        }
      }
    });

    Navigator.push(
      context,
      MaterialPageRoute(
        builder: (context) => ChatScreen(title: name, conversationId: convId, otherUserId: otherUserId),
      ),
    ).then((_) => _scheduleRefresh());
  }

class ChatScreen extends StatefulWidget {

  final String? conversationId;

class _ChatScreenState extends State<ChatScreen> {
  final TextEditingController _textController = TextEditingController();
  final ScrollController _scrollController = ScrollController();
  final List<Message> _messages = [];
  final GlobalKey _inputKey = GlobalKey();
  bool _isAiTyping = false;
  bool _isLoadingMessages = true;
  
  // Long press state
  Message? _selectedMessage;
  int? _selectedMessageIndex;
  List<Message> _selectedAlbumImages = [];
  OverlayEntry? _menuOverlay;

  final supabase = Supabase.instance.client;
  RealtimeChannel? _chatChannel;
  Timer? _pollTimer;
  String? _lastCreatedAt;
  String? _currentUserId;
  String? _resolvedConversationId;

  @override
  void initState() {
    super.initState();
    if (widget.isAi) {
      _messages.add(Message('ai_1', "Hello! I am Dinor AI. Ask me anything and I'll search the web.", false));
      _isLoadingMessages = false;
    } else {
      _currentUserId = supabase.auth.currentUser?.id;
      _resolvedConversationId = widget.conversationId;
      // Paint cached messages before the first frame: no skeleton, no flash.
      final cached = widget.conversationId == null ? null : LocalCache.peekMessages(widget.conversationId!);
      if (cached != null && cached.isNotEmpty) {
        _messages.addAll(cached.map(_rowToMessage));
        _isLoadingMessages = false;
        _jumpToBottom();
      }
      _loadChat();
    }
  }

  Message _rowToMessage(Map<String, dynamic> row) => Message(
        row['id'].toString(),
        (row['content'] ?? '').toString(),
        row['sender_id'] == _currentUserId,
        isImage: row['is_image'] ?? false,
        isRead: row['is_read'] ?? false,
        albumId: row['album_id'] as String?,
        caption: row['caption'] as String?,
      );

  /// Opens the chat: paints cached messages at once, then reconciles with the server.
  Future<void> _loadChat() async {
    // 2. New chats only know the other user's id; resolve the conversation.
    if (_resolvedConversationId == null) {
      if (widget.otherUserId == null) {
        if (mounted) setState(() => _isLoadingMessages = false);
        return;
      }
      try {
        _resolvedConversationId = await ChatService.getOrCreateConversation(widget.otherUserId!);
      } catch (e) {
        debugPrint('Could not open conversation: $e');
        if (mounted) setState(() => _isLoadingMessages = false);
        return;
      }
    }
    final convId = _resolvedConversationId!;

    // 3. Subscribe first, then fetch, so no message can slip between the two.
    _subscribeRealtime(convId);
    // Fallback: check for new messages every few seconds, in case Realtime lags.
    _pollTimer?.cancel();
    _pollTimer = Timer.periodic(const Duration(seconds: 1), (_) => _pollNewMessages());
    await _fetchMessages();
    ChatService.markAsRead(convId);
  }

  /// Realtime inserts for this conversation only (filtered on the server).
  void _subscribeRealtime(String convId) {
    _chatChannel?.unsubscribe();
    _chatChannel = supabase.channel('chat:$convId').onPostgresChanges(
      event: PostgresChangeEvent.all,
      schema: 'public',
      table: 'messages',
      filter: PostgresChangeFilter(
        type: PostgresChangeFilterType.eq,
        column: 'conversation_id',
        value: convId,
      ),
      callback: (payload) {
        if (payload.eventType == PostgresChangeEvent.insert) {
          if (mounted) _ingest(payload.newRecord);
        } else if (payload.eventType == PostgresChangeEvent.update) {
          if (mounted) {
            final updated = payload.newRecord;
            final idx = _messages.indexWhere((m) => m.id == updated['id'].toString());
            if (idx != -1) {
              setState(() {
                _messages[idx] = _rowToMessage(updated);
              });
            }
          }
        }
      },
    ).subscribe();
  }

  /// Adds a server row exactly once. When our own message echoes back over
  /// realtime, it replaces the pending bubble instead of appearing twice.
  void _ingest(Map<String, dynamic> row) {
    final createdAt = row['created_at']?.toString();
    if (createdAt != null && (_lastCreatedAt == null || createdAt.compareTo(_lastCreatedAt!) > 0)) {
      _lastCreatedAt = createdAt;
    }
    final id = row['id'].toString();
    if (_messages.any((m) => m.id == id)) return;

    final mine = row['sender_id'] == _currentUserId;
    final isImage = row['is_image'] ?? false;
    final incoming = _rowToMessage(row);

    setState(() {
      if (mine) {
        final pending = _messages.indexWhere((m) =>
            m.id.startsWith('temp_') && m.status == MessageStatus.sending && m.isMe && m.isImage == isImage);
        if (pending != -1) {
          _messages[pending] = incoming;
          return;
        }
      }
      _messages.add(incoming);
    });
    _scrollToBottom();
    _persist();
  }

  /// Merges the server history with any bubbles still sending or failed.
  Future<void> _fetchMessages() async {
    final convId = _resolvedConversationId;
    if (convId == null) return;
    try {
      final data = await supabase
          .from('messages')
          .select()
          .eq('conversation_id', convId)
          .order('created_at', ascending: true);
      if (!mounted) return;

      final rawRows = List<Map<String, dynamic>>.from(data);
      if (rawRows.isNotEmpty) _lastCreatedAt = rawRows.last['created_at']?.toString();
      final server = rawRows.map(_rowToMessage).toList();
      final serverIds = server.map((m) => m.id).toSet();
      final localPending = _messages.where((m) => m.id.startsWith('temp_') && !serverIds.contains(m.id)).toList();
      final next = [...server, ...localPending];

      // Only rebuild when something actually changed (the common case on reopen is no change).
      if (!_sameMessages(_messages, next)) {
        final wasEmpty = _messages.isEmpty;
        setState(() {
          _messages
            ..clear()
            ..addAll(next);
        });
        _persist();
        if (wasEmpty) {
          _jumpToBottom();
        } else {
          _scrollToBottom();
        }
      }
    } catch (e) {
      debugPrint('Error fetching messages: $e');
    } finally {
      if (mounted && _isLoadingMessages) setState(() => _isLoadingMessages = false);
    }
  }

  bool _sameMessages(List<Message> a, List<Message> b) {
    if (a.length != b.length) return false;
    for (var i = 0; i < a.length; i++) {
      final x = a[i];
      final y = b[i];
      if (x.id != y.id || x.text != y.text || x.isRead != y.isRead ||
          x.status != y.status || x.caption != y.caption) {
        return false;
      }
    }
    return true;
  }

  /// Scrolls to the newest message without animating (used when a chat opens).
  void _jumpToBottom() {
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (mounted && _scrollController.hasClients) {
        _scrollController.jumpTo(_scrollController.position.maxScrollExtent);
      }
    });
  }

  /// Saves confirmed messages to the local cache (never pending or AI bubbles).
  void _persist() {
    final convId = _resolvedConversationId;
    if (convId == null || widget.isAi) return;
    final rows = _messages
        .where((m) => !m.id.startsWith('temp_') && m.id != 'ai_1')
        .map((m) => <String, dynamic>{
              'id': m.id,
              'content': m.text,
              'sender_id': m.isMe ? _currentUserId : 'other',
              'is_image': m.isImage,
              'is_read': m.isRead,
              'album_id': m.albumId,
              'caption': m.caption,
            })
        .toList();
    LocalCache.saveMessages(convId, rows);
  }

  /// Polling fallback: fetches only messages newer than the last one we have.
  Future<void> _pollNewMessages() async {
    final convId = _resolvedConversationId;
    if (convId == null || !mounted) return;
    try {
      final rows = await supabase
          .from('messages')
          .select()
          .eq('conversation_id', convId)
          .gt('created_at', _lastCreatedAt ?? '1970-01-01T00:00:00+00:00')
          .order('created_at', ascending: true);
      for (final row in List<Map<String, dynamic>>.from(rows)) {
        if (!mounted) return;
        _ingest(row);
      }
    } catch (e) {
      debugPrint('message poll: $e');
    }
  }

  @override
  void dispose() {
    _pollTimer?.cancel();
    _menuOverlay?.remove();
    _menuOverlay = null;
    _chatChannel?.unsubscribe();
    _textController.dispose();
    _scrollController.dispose();
    super.dispose();
  }

  Future<void> _sendMessage({String? customText}) async {
    final rawText = customText ?? _textController.text.trim();
    if (rawText.isEmpty) return;

    // Sanitize in C++ (singleton core; Dart fallback if unavailable).
    final text = DinorCore.load().batchSanitize(rawText);
    if (text.isEmpty) return;

    if (customText == null) _textController.clear();

    if (widget.isAi) {
      setState(() => _messages.add(Message(DateTime.now().microsecondsSinceEpoch.toString(), text, true)));
      _scrollToBottom();
      _triggerAiResponse(text);
      return;
    }

    final me = _currentUserId;
    final convId = _resolvedConversationId;
    if (me == null || convId == null) return;

    final tempId = 'temp_${DateTime.now().microsecondsSinceEpoch}';
    setState(() => _messages.add(Message(tempId, text, true, status: MessageStatus.sending)));
    _scrollToBottom();

    try {
      final response = await supabase.from('messages').insert({
        'sender_id': me,
        'content': text,
        'is_image': false,
        'conversation_id': convId,
        'is_read': false,
      }).select('id').single();

      final realId = response['id'].toString();
      if (mounted) {
        setState(() {
          final i = _messages.indexWhere((m) => m.id == tempId);
          if (i != -1) {
            if (_messages.any((m) => m.id == realId)) {
              _messages.removeAt(i); // realtime already added the real row
            } else {
              _messages[i] = Message(realId, text, true, status: MessageStatus.sent);
            }
          }
        });
        _persist();
      }

      try {
        await supabase.from('conversations').update({
          'last_message': text,
          'last_message_time': DateTime.now().toUtc().toIso8601String(),
        }).eq('id', convId);
      } catch (e) {
        debugPrint('Could not update conversation preview: $e');
      }
    } catch (e) {
      debugPrint('Failed to send: $e');
      if (!mounted) return;
      setState(() {
        final i = _messages.indexWhere((m) => m.id == tempId);
        // Keep the user's text so nothing is lost; the bubble shows the failed state.
        if (i != -1) _messages[i] = Message(tempId, text, true, status: MessageStatus.failed);
      });
    }
  }

  void _triggerAiResponse(String userText, {String? imageBase64}) async {
    setState(() => _isAiTyping = true);
    _scrollToBottom();
    try {
      String response = await DinorAI.getResponse(userText, imageBase64: imageBase64);

        _messages.add(Message(DateTime.now().toString(), response, false));
      });
    } catch (e) {
      if (!mounted) return;
      setState(() => _isAiTyping = false);
    }
    _scrollToBottom();
  }

  void _deleteMessage(Message msg) {
    final List<Message> toDelete = msg.albumId != null
        ? _messages.where((m) => m.albumId == msg.albumId).toList()
        : [msg];

    setState(() {
      _messages.removeWhere((m) => toDelete.any((d) => d.id == m.id));
    });
    _persist();

    final convId = _resolvedConversationId;
    for (var m in toDelete) {
      if (m.id.startsWith('temp_')) continue;
      supabase.from('messages').delete().eq('id', m.id).then((_) {
        debugPrint("Deleted message ${m.id} from Supabase");
      }).catchError((e) {
        debugPrint("Failed to delete message ${m.id}: $e");
      });
      if (convId != null) LocalCache.removeMessage(convId, m.id);
    }
  }

class _AnimatedSendingDotsState extends State<_AnimatedSendingDots> with SingleTickerProviderStateMixin {

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 1200),
    )..repeat();
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _controller,
      builder: (context, child) {
        final t = _controller.value;
        return Row(
          mainAxisSize: MainAxisSize.min,
          children: List.generate(3, (index) {
            double scale = 1.0;
            try {
              scale = DinorCore.load().getAnimationScale(t, index);
            } catch (e) {
              double delay = index * 0.2;
              double progress = (t - delay) % 1.0;
              if (progress < 0) progress += 1.0;
              scale = 1.0 + (0.6 * (progress < 0.5 ? progress * 2 : (1 - progress) * 2));
            }
            
            return Padding(
              padding: const EdgeInsets.symmetric(horizontal: 1.5),
              child: Transform.scale(
                scale: scale,
                child: Container(
                  width: 4,
                  height: 4,
                  decoration: const BoxDecoration(
                    color: DinorTheme.textGrey,
                    shape: BoxShape.circle,
                  ),
                ),
              ),
            );
          }),
        );
      },
    );
  }
}

import 'package:flutter/material.dart';
import 'dart:math';

class AnimatedAIAvatar extends StatefulWidget {
  final double size;
  final bool isThinking;

  const AnimatedAIAvatar({
    super.key,
    this.size = 40,
    this.isThinking = false,
  });

  @override
  State<AnimatedAIAvatar> createState() => _AnimatedAIAvatarState();
}

class _AnimatedAIAvatarState extends State<AnimatedAIAvatar> with SingleTickerProviderStateMixin {
  late AnimationController _controller;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: const Duration(seconds: 3),
    );
    // Only animate at 30fps instead of 60fps to stop frame drops
    _controller.repeat(period: const Duration(seconds: 3));
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _controller,
      builder: (context, child) {
        return CustomPaint(
          size: Size(widget.size, widget.size),
          painter: _AIAvatarPainter(
            animationValue: _controller.value,
            isThinking: widget.isThinking,
          ),
        );
      },
    );
  }
}

class _AIAvatarPainter extends CustomPainter {
  final double animationValue;
  final bool isThinking;

  _AIAvatarPainter({required this.animationValue, required this.isThinking});

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final radius = size.width / 2;
    final t = animationValue;

    // 1. Draw the base 3D Orb (Gradient makes it look spherical)
    final orbPaint = Paint()
      ..shader = const RadialGradient(
        colors: [
          Color(0xFF80E5FF), // Bright highlight
          Color(0xFF24A1DE), // Brand blue
          Color(0xFF005A8A), // Dark shadow edge
        ],
        stops: [0.1, 0.6, 1.0],
        center: Alignment(-0.3, -0.3), // Off-center highlight for 3D effect
        radius: 0.9,
      ).createShader(Rect.fromCircle(center: center, radius: radius));

    canvas.drawCircle(center, radius, orbPaint);

    // 2. Draw rotating 3D rings (Orbits)
    final ringPaint = Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2.0
      ..color = Colors.white.withValues(alpha: 0.6);

    final ringPaint2 = Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.5
      ..color = const Color(0xFF24A1DE).withValues(alpha: 0.8);

    canvas.save();
    canvas.translate(center.dx, center.dy);

    // Ring 1 (Tilted)
    canvas.save();
    canvas.rotate(t * 2 * pi);
    canvas.drawOval(
      Rect.fromCenter(center: Offset.zero, width: radius * 1.8, height: radius * 0.6),
      ringPaint,
    );
    canvas.restore();

    // Ring 2 (Different angle)
    canvas.save();
    canvas.rotate(-t * 2 * pi + pi / 4);
    canvas.drawOval(
      Rect.fromCenter(center: Offset.zero, width: radius * 1.6, height: radius * 1.2),
      ringPaint2,
    );
    canvas.restore();

    canvas.restore();

    // 3. Draw a glowing "core" in the center
    final corePulse = isThinking ? (sin(t * 10) * 0.2 + 0.8) : 0.5;
    final corePaint = Paint()
      ..color = Colors.white.withValues(alpha: corePulse)
      ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 4);
    
    canvas.drawCircle(center, radius * 0.3, corePaint);

    // 4. Outer glow if thinking
    if (isThinking) {
      final glowPaint = Paint()
        ..color = const Color(0xFF24A1DE).withValues(alpha: 0.3 * corePulse)
        ..style = PaintingStyle.stroke
        ..strokeWidth = 6
        ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 8);
      canvas.drawCircle(center, radius + 4, glowPaint);
    }
  }

  @override
  bool shouldRepaint(covariant _AIAvatarPainter oldDelegate) {
    return true; // Must be true to rotate the rings, but we reduced the frame rate
  }
}

class CustomMediaPicker extends StatefulWidget {
  const CustomMediaPicker({super.key});

  @override
  State<CustomMediaPicker> createState() => _CustomMediaPickerState();
}

class _CustomMediaPickerState extends State<CustomMediaPicker> {
  List<AssetEntity> _media = [];
  final List<AssetEntity> _selectedMedia = [];
  bool _isLoading = true;
  bool _hasPermission = false;
  final Map<String, Uint8List> _thumbnailCache = {};
  
  final TextEditingController _captionController = TextEditingController();
  final FocusNode _captionFocus = FocusNode();

  @override
  void initState() {
    super.initState();
    _requestPermissionAndLoad();
  }

  @override
  void dispose() {
    _captionController.dispose();
    _captionFocus.dispose();
    super.dispose();
  }

  Future<void> _requestPermissionAndLoad() async {
    final PermissionStatus status = await Permission.photos.request();
    final PermissionStatus videoStatus = await Permission.videos.request();

    if (status.isGranted || videoStatus.isGranted || await Permission.storage.isGranted) {
      setState(() => _hasPermission = true);
      
      // Load directly (Isolate doesn't support photo_manager natively)
      final paths = await PhotoManager.getAssetPathList(
        type: RequestType.all,
        onlyAll: true,
      );
      
      if (paths.isNotEmpty) {
        final assets = await paths.first.getAssetListPaged(page: 0, size: 100);
        setState(() {
          _media = assets;
          _isLoading = false;
        });
        _preloadThumbnails(assets); // Re-added the caching call!
      } else {
        setState(() => _isLoading = false);
      }

      // Pre-load thumbnails smoothly
    } else {
      setState(() {
        _isLoading = false;
        _hasPermission = false;
      });
    }
  }

  Future<void> _preloadThumbnails(List<AssetEntity> assets) async {
    // Load in smaller batches to keep the UI thread responsive
    for (int i = 0; i < assets.length; i++) {
      final asset = assets[i];
      final thumbData = await asset.thumbnailDataWithSize(const ThumbnailSize(150, 150));
      if (thumbData != null) {
        _thumbnailCache[asset.id] = thumbData;
      }
      
      // Update the UI every 5 images instead of every single image
      if (i % 5 == 0 && mounted) {
        setState(() {});
      }
    }
    if (mounted) setState(() {}); // Final update
  }

  void _toggleSelection(AssetEntity asset) {
    setState(() {
      if (_selectedMedia.contains(asset)) {
        _selectedMedia.remove(asset);
      } else {
        _selectedMedia.add(asset);
      }
    });
  }

  void _sendMedia() {
    // Return the selected media and the caption back to ChatScreen
    Navigator.pop(context, {
      'media': _selectedMedia,
      'caption': _captionController.text.trim(),
    });
  }

  @override
  Widget build(BuildContext context) {
    // We use a Column aligned to the bottom so the keyboard naturally pushes everything up
    return DraggableScrollableSheet(
      initialChildSize: 0.5,
      minChildSize: 0.4,
      maxChildSize: 0.95,
      snap: true,
      snapSizes: const [0.5, 0.95],

        return Container(

            color: DinorTheme.surfaceDark,
            borderRadius: BorderRadius.vertical(top: Radius.circular(24)),
          ),
          // Padding uses MediaQuery.viewInsets.bottom so the caption bar rides the keyboard
          child: Padding(
            padding: EdgeInsets.only(bottom: MediaQuery.of(context).viewInsets.bottom),
            child: Column(
              children: [
                // 1. Drag Handle
                Center(
                  child: Container(
                    margin: const EdgeInsets.symmetric(vertical: 12),
                    width: 40,
                    height: 4,
                    decoration: BoxDecoration(
                      color: DinorTheme.textGrey.withValues(alpha: 0.3),
                      borderRadius: BorderRadius.circular(2),
                    ),
                  ),
                ),
                
                // 2. Header (Title + Done Button)
                Padding(
                  padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 8),
                  child: Row(
                    mainAxisAlignment: MainAxisAlignment.spaceBetween,
                    children: [
                      const Text(
                        'Gallery',
                        style: TextStyle(fontWeight: FontWeight.bold, fontSize: 18, color: Colors.white),
                      ),
                      if (_selectedMedia.isNotEmpty)
                        ElevatedButton(
                          onPressed: () {
                            // If they have selections, focus the caption to let them type
                            _captionFocus.requestFocus();
                          },
                          style: ElevatedButton.styleFrom(
                            backgroundColor: const Color(0xFF229ED9),
                            foregroundColor: Colors.white,
                          ),
                          child: Text('Done (${_selectedMedia.length})'),
                        ),
                    ],
                  ),
                ),

                // 3. Media Grid
                Expanded(
                  child: _isLoading
                      ? const Center(child: CircularProgressIndicator())
                      : !_hasPermission
                          ? Center(
                              child: Column(
                                mainAxisAlignment: MainAxisAlignment.center,
                                children: [
                                  const Icon(Icons.lock, size: 48, color: Colors.grey),
                                  const SizedBox(height: 16),
                                  const Text("Permission needed to view photos."),
                                  TextButton(
                                    onPressed: _requestPermissionAndLoad,
                                    child: const Text("Grant Access"),
                                  ),
                                ],
                              ),
                            )
                          : GridView.builder(
                              controller: scrollController,
                              padding: const EdgeInsets.all(8),
                              gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
                                crossAxisCount: 3,
                                crossAxisSpacing: 4,
                                mainAxisSpacing: 4,
                              ),
                              itemCount: _media.length,
                              itemBuilder: (context, index) {
                                final asset = _media[index];
                                final isSelected = _selectedMedia.contains(asset);

                                return GestureDetector(
                                  onTap: () => _toggleSelection(asset),
                                  child: Stack(
                                    fit: StackFit.expand,
                                    children: [
                                      // Image Thumbnail
                                      Container(
                                        decoration: BoxDecoration(
                                          borderRadius: BorderRadius.circular(8),
                                          color: DinorTheme.bgDark,
                                        ),
                                        clipBehavior: Clip.hardEdge,
                                        child: _thumbnailCache.containsKey(asset.id)
                                            ? Image.memory(
                                                _thumbnailCache[asset.id]!, 
                                                fit: BoxFit.cover,
                                                cacheWidth: 150, // Decode at thumbnail size
                                              )
                                            : const Center(child: SizedBox(width: 20, height: 20, child: CircularProgressIndicator(strokeWidth: 2))),
                                      ),
                                      
                                      // Selection Overlay
                                      if (isSelected)
                                        Container(
                                          decoration: BoxDecoration(
                                            color: const Color(0xFF229ED9).withValues(alpha: 0.4),
                                            borderRadius: BorderRadius.circular(8),
                                            border: Border.all(color: const Color(0xFF229ED9), width: 3),
                                          ),
                                          child: Center(
                                            child: CircleAvatar(
                                              radius: 12,
                                              backgroundColor: const Color(0xFF229ED9),
                                              child: Text(
                                                _selectedMedia.indexOf(asset).toString(),
                                                style: const TextStyle(color: Colors.white, fontSize: 12),
                                              ),
                                            ),
                                          ),
                                        ),
                                    ],
                                  ),
                                );
                              },
                            ),
                ),

                // 4. Caption Composer (Only shows when media is selected)
                AnimatedContainer(
                  duration: const Duration(milliseconds: 250),
                  curve: Curves.easeInOut,
                  height: _selectedMedia.isNotEmpty ? 60 : 0,
                  decoration: BoxDecoration(
                    color: DinorTheme.surfaceDark,
                    border: Border(top: BorderSide(color: DinorTheme.bgDark)),
                  ),
                  child: _selectedMedia.isNotEmpty
                      ? Row(
                          children: [
                            // Preview of the first selected photo
                            Padding(
                              padding: const EdgeInsets.all(8.0),
                              child: Container(
                                width: 40,
                                height: 40,
                                decoration: BoxDecoration(
                                  borderRadius: BorderRadius.circular(6),
                                  color: DinorTheme.bgDark,
                                ),
                                clipBehavior: Clip.hardEdge,
                                child: _thumbnailCache.containsKey(_selectedMedia.first.id)
                                    ? Image.memory(
                                        _thumbnailCache[_selectedMedia.first.id]!, 
                                        fit: BoxFit.cover,
                                        cacheWidth: 80,
                                      )
                                    : const SizedBox(),
                              ),
                            ),
                            // Caption Input
                            Expanded(
                              child: TextField(
                                controller: _captionController,
                                focusNode: _captionFocus,
                                style: const TextStyle(color: Colors.white),
                                decoration: const InputDecoration(
                                  hintText: 'Add a caption...',
                                  hintStyle: TextStyle(color: DinorTheme.textGrey),
                                  border: InputBorder.none,
                                  contentPadding: EdgeInsets.symmetric(horizontal: 8),
                                ),
                              ),
                            ),
                            // Send Button
                            IconButton(
                              icon: const Icon(Icons.send, color: Color(0xFF229ED9)),
                              onPressed: _sendMedia,
                            ),
                          ],
                        )
                      : null,
                ),
              ],
            ),
          ),
        );
      },
    );
  }

}

import 'package:flutter/material.dart';

class AnimatedEmoji extends StatefulWidget {
  final String emoji;
  final double size;
  final VoidCallback? onTap; // New callback

  const AnimatedEmoji({super.key, required this.emoji, this.size = 40, this.onTap});

  @override
  State<AnimatedEmoji> createState() => _AnimatedEmojiState();
}

class _AnimatedEmojiState extends State<AnimatedEmoji> with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  late Animation<double> _scaleAnimation;
  late Animation<double> _rotateAnimation;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 400),
    );

    // Springy scale animation
    _scaleAnimation = TweenSequence<double>([
      TweenSequenceItem(tween: Tween(begin: 1.0, end: 1.5).chain(CurveTween(curve: Curves.easeOutBack)), weight: 50),
      TweenSequenceItem(tween: Tween(begin: 1.5, end: 1.0).chain(CurveTween(curve: Curves.easeInOut)), weight: 50),
    ]).animate(_controller);

    // Subtle 3D rotation
    _rotateAnimation = Tween<double>(begin: 0, end: 0.1).animate(
      CurvedAnimation(parent: _controller, curve: Curves.easeInOut),
    );
  }

  void playAnimation() {
    _controller.forward(from: 0);
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return GestureDetector(
      onTap: () {
        playAnimation();
        if (widget.onTap != null) {
          widget.onTap!();
        }
      },
      child: AnimatedBuilder(
        animation: _controller,
        builder: (context, child) {
          return Transform(
            transform: Matrix4.identity()
              ..scaleByDouble(_scaleAnimation.value, _scaleAnimation.value, _scaleAnimation.value, 1.0)
              ..rotateZ(_rotateAnimation.value),
            alignment: Alignment.center,
            child: Text(
              widget.emoji,
              style: TextStyle(
                fontSize: widget.size,
                // Crisp text rendering for emojis
                fontFamily: 'Noto Color Emoji', 
              ),
            ),
          );
        },
      ),
    );
  }
}

  @override

  Future<void> _fetchGifs({String? query}) async {
    // Check internet first
    final connectivityResult = await Connectivity().checkConnectivity();
    final isOffline = connectivityResult.contains(ConnectivityResult.none);

    if (isOffline) {
      setState(() {
        _hasInternet = false;
        _isGifLoading = false;
        _gifs = [];
      });
      return;
    } else {
      setState(() {
        _hasInternet = true;
        _isGifLoading = true;
      });
    }

    final endpoint = query == null || query.isEmpty
        ? 'https://api.giphy.com/v1/gifs/trending?api_key=$_apiKey&limit=20&rating=g'
        : 'https://api.giphy.com/v1/gifs/search?api_key=$_apiKey&q=$query&limit=20&rating=g';

    try {
      // Giphy requires a User-Agent header to prevent bot blocking
      final response = await http.get(
        Uri.parse(endpoint),
        headers: {
          'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)',
          'Accept': 'application/json',
        },
      );

      if (response.statusCode == 200) {
        final data = jsonDecode(response.body);
        final List<dynamic> fetchedGifs = data['data'] ?? [];
        
        setState(() {
          _gifs = fetchedGifs;
          _hasInternet = true;
        });
        
        // If Giphy returns 0 results, it means the API key is invalid or rate-limited
        if (fetchedGifs.isEmpty) {
          debugPrint("Giphy API returned 0 results. Check your API key.");
        }
      } else {
        debugPrint("Giphy API Error: Status ${response.statusCode} - ${response.body}");
      }
    } catch (e) {
      debugPrint("Giphy network error: $e");
    } finally {
      setState(() => _isGifLoading = false);
    }
  }

    if (_searchQuery == val) return;
    setState(() => _searchQuery = val);
    if (_currentTab == 1) {
      _fetchGifs(query: _searchQuery);
    }
  }

import 'package:flutter/material.dart';

class LoginScreen extends StatefulWidget {
  const LoginScreen({super.key});

  @override
  State<LoginScreen> createState() => _LoginScreenState();
}

class _LoginScreenState extends State<LoginScreen> {
  final _emailController = TextEditingController();
  final _codeController = TextEditingController();
  final _supabase = Supabase.instance.client;

  bool _isLoading = false;
  bool _codeSent = false;
  String? _errorMessage;
  String? _statusMessage;

  static const Color brandBlue = Color(0xFF229ED9);

  Future<void> _requestCode() async {
    final email = _emailController.text.trim().toLowerCase();

    if (email.isEmpty || !RegExp(r'^[^\s@]+@[^\s@]+\.[^\s@]+$').hasMatch(email)) {
      setState(() {
        _errorMessage = 'Enter a valid email address.';
        _statusMessage = null;
      });
      return;
    }

    FocusScope.of(context).unfocus();
    setState(() {
      _isLoading = true;
      _errorMessage = null;
      _statusMessage = null;
    });

    try {
      final response = await _supabase.functions.invoke(
        'request-email-code',
        body: {'email': email},
      );

      final data = response.data;
      if (response.status < 200 || response.status >= 300) {
        throw Exception(
          data is Map && data['error'] is String
              ? data['error']
              : 'Could not request a code. Please try again.',
        );
      }

      if (!mounted) return;
      setState(() {
        _codeSent = true;
        _statusMessage = data is Map && data['message'] is String
            ? data['message']
            : 'Check your email for a sign-in code.';
      });
    } on FunctionException catch (e) {
      if (!mounted) return;
      final details = e.details;
      setState(() {
        _errorMessage = details is Map && details['error'] is String
            ? details['error'] as String
            : 'Could not send the code. Check your connection and try again.';
      });
    } catch (e) {
      if (!mounted) return;
      setState(() => _errorMessage = e.toString().replaceFirst('Exception: ', ''));
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

  Future<void> _verifyCode() async {
    final email = _emailController.text.trim().toLowerCase();
    final code = _codeController.text.trim();

    if (!RegExp(r'^\d{8}$').hasMatch(code)) {
      setState(() => _errorMessage = 'Enter the eight-digit code from your email.');
      return;
    }

    FocusScope.of(context).unfocus();
    setState(() {
      _isLoading = true;
      _errorMessage = null;
      _statusMessage = null;
    });

    try {
      final response = await _supabase.functions.invoke(
        'verify-email-code',
        body: {'email': email, 'code': code},
      );

      final data = response.data;
      if (response.status < 200 ||
          response.status >= 300 ||
          data is! Map ||
          data['refresh_token'] is! String ||
          (data['refresh_token'] as String).isEmpty) {
        throw Exception(
          data is Map && data['error'] is String
              ? data['error']
              : 'The code is invalid or expired. Request a new code.',
        );
      }

      // Establish a persistent, real Supabase Auth session.
      await _supabase.auth.setSession(data['refresh_token'] as String);

      // Automatically create a profile for the user if it doesn't exist
      final user = _supabase.auth.currentUser;
      if (user != null) {
        final existingProfile = await _supabase.from('profiles').select().eq('id', user.id).maybeSingle();
        if (existingProfile == null) {
          // Use the email prefix as the default username (e.g., 'john' from 'john@gmail.com')
          final defaultUsername = email.split('@')[0];
          await _supabase.from('profiles').insert({
            'id': user.id,
            'username': defaultUsername,
          });
        }
      }

      if (!mounted) return;
      Navigator.of(context).pushAndRemoveUntil(
        MaterialPageRoute(builder: (_) => const HomeScreen()),
        (route) => false,
      );
    } on FunctionException catch (e) {
      if (!mounted) return;
      final details = e.details;
      setState(() {
        _errorMessage = details is Map && details['error'] is String
            ? details['error'] as String
            : 'Could not verify the code. Please try again.';
      });
    } catch (e) {
      if (!mounted) return;
      setState(() => _errorMessage = e.toString().replaceFirst('Exception: ', ''));
    } finally {
      if (mounted) setState(() => _isLoading = false);
    }
  }

    _emailController.dispose();
    _codeController.dispose();
    super.dispose();
  }

            child: ConstrainedBox(

              child: Column(

                mainAxisSize: MainAxisSize.min,
                children: [

                      fontWeight: FontWeight.bold,
                    ),
                  ),
                  const SizedBox(height: 8),
                  const Text(

                    textAlign: TextAlign.center,
                    style: TextStyle(
                      color: DinorTheme.textGrey,
                      fontSize: 15,
                    ),
                  ),
                  const SizedBox(height: 32),
                  TextField(
                    controller: _emailController,
                    enabled: !_isLoading && !_codeSent,
                    keyboardType: TextInputType.emailAddress,
                    textInputAction: TextInputAction.done,
                    autocorrect: false,
                    autofillHints: const [AutofillHints.email],
                    decoration: InputDecoration(

                      prefixIcon: const Icon(Icons.email_outlined),
                      filled: true,
                      fillColor: DinorTheme.surfaceDark,

class DinorAI {
  static Future<String> getResponse(String userMessage, {String? imageBase64}) async {
    final connectivityResult = await Connectivity().checkConnectivity();
    if (connectivityResult.contains(ConnectivityResult.none)) {
      return "I am currently offline. Please check your internet connection.";
    }

    final supabase = Supabase.instance.client;

    try {
      final response = await supabase.functions.invoke(
        'chat-ai',
        body: {
          'userMessage': userMessage,
          if (imageBase64 != null) 'imageBase64': imageBase64,
        },
      );

      // Safely extract the reply
      if (response.status == 200 && response.data != null) {
        final data = response.data;
        
        // If it's a Map, look for the 'reply' key
        if (data is Map && data['reply'] != null) {
          return data['reply'].toString();
        } 
        // If it's a String, just return it
        else if (data is String) {
          return data;
        }
      }
      
      return "I couldn't generate a response. Please try again. (Status: ${response.status})";
    } catch (e) {
      return "Connection error: ${e.toString()}";
    }
  }
}

class ChatService {

  static Future<List<Map<String, dynamic>>> searchUsers(String query) async {
    final me = myId;
    if (me == null) return [];
    // Escape % and _ so user input cannot act as wildcards.
    final safe = DinorCore.load().escapeLike(query.trim());
    if (safe.isEmpty) return [];
    final response = await supabase
        .from('profiles')
        .select()
        .ilike('username', '%$safe%')
        .neq('id', me)
        .limit(10);
    return List<Map<String, dynamic>>.from(response);
  }

  static Future<String> getOrCreateConversation(String otherUserId) async {
    final myId = supabase.auth.currentUser!.id;

    final existing = await supabase
        .from('conversations')
        .select('id')
        .or('and(user1_id.eq.$myId,user2_id.eq.$otherUserId),and(user1_id.eq.$otherUserId,user2_id.eq.$myId)')
        .maybeSingle();

    if (existing != null) return existing['id'] as String;

    final newConv = await supabase
        .from('conversations')
        .insert({'user1_id': myId, 'user2_id': otherUserId})
        .select('id')
        .single();

    return newConv['id'] as String;
  }

  /// Conversations with the other user's profile and unread info.
  /// Unread lookups run in parallel (one round trip of latency, not N).
  static Future<List<Map<String, dynamic>>> getMyConversations() async {
    final myId = supabase.auth.currentUser!.id;

    final response = await supabase
        .from('conversations')
        .select('*, user1:profiles!conversations_user1_id_fkey(*), user2:profiles!conversations_user2_id_fkey(*)')
        .or('user1_id.eq.$myId,user2_id.eq.$myId')
        .order('last_message_time', ascending: false);

    final conversations = List<Map<String, dynamic>>.from(response);

    final unreadLists = await Future.wait(conversations.map((conv) {
      return supabase
          .from('messages')
          .select('content, is_image')
          .eq('conversation_id', conv['id'])
          .neq('sender_id', myId)
          .eq('is_read', false)
          .order('created_at', ascending: false)
          .then((rows) => List<Map<String, dynamic>>.from(rows))
          .catchError((_) => <Map<String, dynamic>>[]);
    }));

    for (var i = 0; i < conversations.length; i++) {
      final unread = unreadLists[i];
      conversations[i]['unread_count'] = unread.length;
      conversations[i]['latest_unread'] = unread.isNotEmpty ? unread.first : null;
    }

    return conversations;
  }

    final me = myId;
    if (me == null) return;
    await supabase
        .from('messages')
        .update({'is_read': true})
        .eq('conversation_id', conversationId)
        .neq('sender_id', me)
        .eq('is_read', false);
  }

import 'dart:convert';

import 'package:shared_preferences/shared_preferences.dart';

import '../ffi/dinor_bindings.dart';

/// Local cache for conversations and chat messages.
///
/// Strategy ("show first, refresh after"):
///  * Startup paints the cached conversation list immediately (no spinner).
///  * Opening a chat paints its cached messages immediately.
///  * Every entry carries `savedAt`; stale entries still paint, then the
///    screen refreshes from the network in the background.
///  * Data is mirrored in memory (instant) and in SharedPreferences (survives restarts).
///  * Message lists are capped so the cache never grows without bound.
class LocalCache {
  LocalCache._();

  static const String _conversationsKey = 'cache.v2.conversations';
  static const String _messagesPrefix = 'cache.v2.messages.';
  static const String _indexKey = 'cache.v2.message_index';

  /// Entries older than this are refreshed in the background.
  static const Duration ttl = Duration(minutes: 5);

  /// Max messages kept per conversation on disk and in memory.
  static const int maxMessagesPerConversation = 100;

  /// Max conversations whose messages we keep (least recently used are dropped).
  static const int maxCachedConversations = 30;

  static SharedPreferences? _prefs;
  static bool _loaded = false;

  static final Map<String, _Entry> _memory = {};

  static Future<SharedPreferences> _p() async => _prefs ??= await SharedPreferences.getInstance();

  /// Loads the conversation list into memory. Call once at startup, before runApp
  /// finishes its first frame where possible.
  static Future<void> preload() async {
    if (_loaded) return;
    _loaded = true;
    final prefs = await _p();
    final raw = prefs.getString(_conversationsKey);
    if (raw != null) {
      final entry = _Entry.decode(raw);
      if (entry != null) _memory[_conversationsKey] = entry;
    }
    // Load recent chats into memory now, so opening one paints on the first frame.
    for (final id in prefs.getStringList(_indexKey) ?? const <String>[]) {
      final rawMessages = prefs.getString(_messagesPrefix + id);
      if (rawMessages == null) continue;
      final entry = _Entry.decode(rawMessages);
      if (entry != null) _memory[_messagesPrefix + id] = entry;
    }
  }

  /// Synchronous: cached messages for a chat, or null. Call after [preload].
  static List<Map<String, dynamic>>? peekMessages(String conversationId) {
    final e = _memory[_messagesPrefix + conversationId];
    if (e == null) return null;
    return (e.data as List).map((x) => Map<String, dynamic>.from(x as Map)).toList();
  }

  // ---- conversations -------------------------------------------------------
  /// Synchronous: returns the last saved conversation list, or null.
  static List<Map<String, dynamic>>? peekConversations() {
    final e = _memory[_conversationsKey];
    if (e == null) return null;
    return (e.data as List).map((x) => Map<String, dynamic>.from(x as Map)).toList();
  }

  static bool conversationsAreFresh() => _isFresh(_memory[_conversationsKey]);

  static Future<void> saveConversations(List<Map<String, dynamic>> conversations) async {
    // Only keep the lightweight fields the list actually renders.
    final light = conversations.map((c) => _lightConversation(c)).toList();
    _memory[_conversationsKey] = _Entry(DateTime.now().millisecondsSinceEpoch, light);
    final prefs = await _p();
    await prefs.setString(_conversationsKey, _memory[_conversationsKey]!.encode());
  }

  // ---- messages ------------------------------------------------------------
  /// Loads cached rows for a conversation (memory first, then disk).
  static Future<List<Map<String, dynamic>>?> loadMessages(String conversationId) async {
    final key = _messagesPrefix + conversationId;
    var entry = _memory[key];
    if (entry == null) {
      final prefs = await _p();
      final raw = prefs.getString(key);
      if (raw != null) {
        entry = _Entry.decode(raw);
        if (entry != null) _memory[key] = entry;
      }
    }
    if (entry == null) return null;
    return (entry.data as List).map((x) => Map<String, dynamic>.from(x as Map)).toList();
  }

  static bool messagesAreFresh(String conversationId) =>
      _isFresh(_memory[_messagesPrefix + conversationId]);

  static Future<void> saveMessages(String conversationId, List<Map<String, dynamic>> rows) async {
    final key = _messagesPrefix + conversationId;
    final trimmed = rows.length > maxMessagesPerConversation
        ? rows.sublist(rows.length - maxMessagesPerConversation)
        : rows;
    final entry = _Entry(DateTime.now().millisecondsSinceEpoch, trimmed);
    _memory[key] = entry;

    final prefs = await _p();
    await prefs.setString(key, entry.encode());

    // Track recency so we can evict old conversations.
    final index = List<String>.from(prefs.getStringList(_indexKey) ?? const []);
    index.remove(conversationId);
    index.add(conversationId);
    while (index.length > maxCachedConversations) {
      final evicted = index.removeAt(0);
      _memory.remove(_messagesPrefix + evicted);
      await prefs.remove(_messagesPrefix + evicted);
    }
    await prefs.setStringList(_indexKey, index);
  }

  /// Removes one message from the cached list (keeps cache in sync after a delete).
  static Future<void> removeMessage(String conversationId, String messageId) async {
    final rows = await loadMessages(conversationId);
    if (rows == null) return;
    rows.removeWhere((r) => r['id']?.toString() == messageId);
    await saveMessages(conversationId, rows);
  }

  /// Clears everything (call on sign-out so the next user never sees old data).
  static Future<void> clearAll() async {
    _memory.clear();
    final prefs = await _p();
    final keys = prefs.getKeys().where((k) => k.startsWith('cache.v2.')).toList();
    for (final k in keys) {
      await prefs.remove(k);
    }
  }

  // ---- helpers -------------------------------------------------------------
  static bool _isFresh(_Entry? e) {
    if (e == null) return false;
    return !DinorCore.load().isStale(savedAtMs: e.savedAt, ttlMs: ttl.inMilliseconds);
  }

  static Map<String, dynamic> _lightConversation(Map<String, dynamic> c) {
    Map<String, dynamic>? slim(dynamic p) => p == null
        ? null
        : {'id': (p as Map)['id'], 'username': p['username'], 'avatar_url': p['avatar_url']};
    return {
      'id': c['id'],
      'user1_id': c['user1_id'],
      'user2_id': c['user2_id'],
      'last_message': c['last_message'],
      'last_message_time': c['last_message_time'],
      'unread_count': c['unread_count'] ?? 0,
      'latest_unread': c['latest_unread'],
      'user1': slim(c['user1']),
      'user2': slim(c['user2']),
    };
  }
}

class _Entry {
  final int savedAt;
  final dynamic data;
  _Entry(this.savedAt, this.data);

  String encode() => jsonEncode({'savedAt': savedAt, 'data': data});

  static _Entry? decode(String raw) {
    try {
      final m = jsonDecode(raw) as Map<String, dynamic>;
      return _Entry(m['savedAt'] as int, m['data']);
    } catch (_) {
      return null; // corrupt cache: ignore it
    }
  }
}

#include "dinor_core.h"

#include <algorithm>
#include <cctype>
#include <vector>
#include <chrono>
#include <cmath>
#include <cstring>
#include <string>

namespace {

// Copy `s` into out (size out_len). Returns bytes written or -1 if it does not fit.
int copy_out(const std::string& s, char* out, int out_len) {
    if (!out || out_len <= 0) return -1;
    if (static_cast<long long>(s.size()) + 1 > out_len) return -1;
    std::memcpy(out, s.data(), s.size());
    out[s.size()] = '\0';
    return static_cast<int>(s.size());
}

// Length of the longest prefix of s that is <= max_bytes and does not split a UTF-8 code point.
size_t utf8_boundary(const std::string& s, size_t max_bytes) {
    if (s.size() <= max_bytes) return s.size();
    size_t cut = max_bytes;
    // Walk back over continuation bytes (10xxxxxx) so we land on a code point start.
    while (cut > 0 && (static_cast<unsigned char>(s[cut]) & 0xC0) == 0x80) --cut;
    return cut;
}

// True for control characters that never belong in chat text.
bool is_bad_control(unsigned char c) {
    return (c < 0x20 && c != '\n') || c == 0x7F;
}

std::string lower_ascii(std::string s) {
    for (auto& ch : s) {
        unsigned char c = static_cast<unsigned char>(ch);
        if (c >= 'A' && c <= 'Z') ch = static_cast<char>(c - 'A' + 'a');
    }
    return s;
}

std::string sanitize_keep_newlines(const char* input) {
    std::string out;
    out.reserve(std::strlen(input));
    for (const char* p = input; *p; ++p) {
        if (!is_bad_control(static_cast<unsigned char>(*p))) out.push_back(*p);
    }
    return out;
}

}  // namespace

extern "C" {

const char* dinor_version(void) {
    return "Dinor 0.6.0 (Core)";
}

int dinor_utf8_length(const char* text) {
    if (!text) return -1;
    return static_cast<int>(std::strlen(text));
}

int dinor_encrypt(const char* plaintext, char* out, int out_len) {
    if (!plaintext) return -1;
    std::string s(plaintext);
    std::reverse(s.begin(), s.end());
    return copy_out(s, out, out_len);
}

int dinor_fast_search(const char* haystack, const char* needle) {
    if (!haystack || !needle) return 0;
    if (!*needle) return 1;
    return lower_ascii(haystack).find(lower_ascii(needle)) != std::string::npos ? 1 : 0;
}

int dinor_format_message(const char* input, char* out, int out_len) {
    if (!input) return -1;
    std::string result;
    bool last_was_space = true;
    for (const char* p = input; *p; ++p) {
        if (*p == ' ') {
            if (!last_was_space) { result.push_back(' '); last_was_space = true; }
        } else {
            result.push_back(*p);
            last_was_space = false;
        }
    }
    if (!result.empty() && result.back() == ' ') result.pop_back();
    return copy_out(result, out, out_len);
}

int dinor_extract_web_answer(const char* query, const char* html_text, char* out, int out_len) {
    if (!query || !html_text) return -1;

    std::vector<std::string> keywords;
    {
        std::string q(query);
        size_t i = 0;
        while (i < q.size()) {
            while (i < q.size() && std::isspace(static_cast<unsigned char>(q[i]))) ++i;
            size_t start = i;
            while (i < q.size() && !std::isspace(static_cast<unsigned char>(q[i]))) ++i;
            std::string w;
            for (size_t k = start; k < i; ++k) {
                if (!std::ispunct(static_cast<unsigned char>(q[k]))) w.push_back(q[k]);
            }
            if (w.size() > 2) keywords.push_back(lower_ascii(w));
        }
    }

    std::string best;
    int max_matches = 0;
    std::string text(html_text);
    size_t start = 0;
    while (start <= text.size()) {
        size_t end = text.find('.', start);
        if (end == std::string::npos) end = text.size();
        std::string sentence = text.substr(start, end - start);
        start = end + 1;

        size_t a = sentence.find_first_not_of(" \n\r\t");
        if (a == std::string::npos) continue;
        size_t b = sentence.find_last_not_of(" \n\r\t");
        sentence = sentence.substr(a, b - a + 1);
        if (sentence.size() < 20 || sentence.size() > 300) continue;

        std::string lower = lower_ascii(sentence);
        int matches = 0;
        for (const auto& kw : keywords) {
            if (lower.find(kw) != std::string::npos) matches++;
        }
        if (matches > max_matches) {
            max_matches = matches;
            best = sentence + ".";
        }
    }

    if (best.empty()) return -1;
    return copy_out(best, out, out_len);
}

int dinor_sanitize_input(const char* input, char* out, int out_len) {
    if (!input || !out || out_len <= 0) return -1;
    std::string clean = sanitize_keep_newlines(input);
    clean = clean.substr(0, utf8_boundary(clean, static_cast<size_t>(out_len - 1)));
    return copy_out(clean, out, out_len);
}

int dinor_batch_sanitize(const char* input, char* out, int out_len) {
    if (!input || !out || out_len <= 0) return -1;
    std::string result;
    result.reserve(std::strlen(input));
    bool last_was_space = true;
    for (const char* p = input; *p; ++p) {
        unsigned char c = static_cast<unsigned char>(*p);
        if (is_bad_control(c) || c == '\n' || c == '\r' || c == '\t') {
            // Treat line breaks and tabs as a single space; drop other controls.
            if ((c == '\n' || c == '\r' || c == '\t') && !last_was_space) {
                result.push_back(' ');
                last_was_space = true;
            }
            continue;
        }
        if (c == ' ') {
            if (!last_was_space) { result.push_back(' '); last_was_space = true; }
        } else {
            result.push_back(*p);
            last_was_space = false;
        }
    }
    while (!result.empty() && result.back() == ' ') result.pop_back();
    result = result.substr(0, utf8_boundary(result, static_cast<size_t>(out_len - 1)));
    return copy_out(result, out, out_len);
}

int dinor_escape_like(const char* input, char* out, int out_len) {
    if (!input) return -1;
    std::string r;
    for (const char* p = input; *p; ++p) {
        if (*p == '%' || *p == '_' || *p == '\\') r.push_back('\\');
        r.push_back(*p);
    }
    return copy_out(r, out, out_len);
}

int dinor_calculate_hash(const char* text) {
    if (!text) return 0;
    unsigned long hash = 5381;
    for (const unsigned char* p = reinterpret_cast<const unsigned char*>(text); *p; ++p) {
        hash = ((hash << 5) + hash) + *p;
    }
    return static_cast<int>(hash & 0x7FFFFFFFUL);
}

long long dinor_fast_timestamp(void) {
    using namespace std::chrono;
    return duration_cast<milliseconds>(system_clock::now().time_since_epoch()).count();
}

int dinor_is_stale(long long saved_at_ms, long long ttl_ms, long long now_ms) {
    if (saved_at_ms <= 0 || ttl_ms <= 0) return 1;      // never saved / no TTL: treat as stale
    if (now_ms < saved_at_ms) return 1;                  // clock went backwards: refresh
    return (now_ms - saved_at_ms) > ttl_ms ? 1 : 0;
}

long long dinor_retry_delay_ms(int attempt, long long base_ms, long long cap_ms) {
    if (attempt < 0) attempt = 0;
    if (base_ms <= 0) base_ms = 250;
    if (cap_ms < base_ms) cap_ms = base_ms;
    double d = static_cast<double>(base_ms) * std::pow(2.0, std::min(attempt, 30));
    return static_cast<long long>(std::min(d, static_cast<double>(cap_ms)));
}

float dinor_animation_scale(double time_seconds, int dot_index) {
    double delay = dot_index * 0.2;
    double progress = std::fmod(time_seconds - delay, 1.0);
    if (progress < 0) progress += 1.0;
    double scale = 1.0 + (0.6 * (progress < 0.5 ? progress * 2 : (1 - progress) * 2));
    return static_cast<float>(scale);
}

const char* dinor_get_status_icon(int status_code) {
    switch (status_code) {
        case 0: return "sending";
        case 1: return "sent";
        case 2: return "failed";
        default: return "unknown";
    }
}

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

async function searchGoogleNews(query: string): Promise<string> {
  try {
    const response = await fetch(`https://news.google.com/rss/search?q=${encodeURIComponent(query)}&hl=en-US&gl=US&ceid=US:en`);
    const xml = await response.text();
    const items: string[] = [];
    const itemBlocks = xml.split('<item>');
    for (let i = 1; i < itemBlocks.length && items.length < 5; i++) {
      const titleMatch = itemBlocks[i].match(/<title>(.*?)<\/title>/);
      if (titleMatch && titleMatch[1]) {
        items.push(`- ${titleMatch[1].replace(/<[^>]+>/g, '').trim()}`);
      }
    }
    return items.join('\n');
  } catch (e) { return ""; }
}

serve(async (req) => {
  if (req.method === 'OPTIONS') return new Response('ok', { headers: corsHeaders })

  try {

      headers: { 'Authorization': `Bearer ${groqApiKey}` }

    const availableModels: string[] = modelsData.data?.map((m: any) => m.id) || [];

    // -------------------------

    if (imageBase64) {

      }

      const visionRes = await fetch('https://api.groq.com/openai/v1/chat/completions', {
        method: 'POST',

        body: JSON.stringify({

            }

      const visionData = await visionRes.json();
      if (visionData.choices && visionData.choices.length > 0) {

    return new Response(JSON.stringify({ reply: `AI Error: ${error.message}` }), {
      status: 200,
      headers: { ...corsHeaders, 'Content-Type': 'application/json' },
    })
  }
})

const COOKIE = "dinor_oauth_state";

function page(message: string, status = 200, clearCookie = false) {
  const headers = new Headers({
    "Content-Type": "text/html; charset=utf-8",
    "Cache-Control": "no-store",
    "Content-Security-Policy": "default-src 'none'; style-src 'unsafe-inline'",
    "X-Content-Type-Options": "nosniff",
    "Referrer-Policy": "no-referrer",
  });

  if (clearCookie) {
    headers.append(
      "Set-Cookie",
      `${COOKIE}=; HttpOnly; Secure; SameSite=Lax; Path=/functions/v1/google-oauth-callback; Max-Age=0`,
    );
  }

  return new Response(
    `<!doctype html><html><head><meta name="viewport" content="width=device-width,initial-scale=1"></head><body><h2>Dinor Google setup</h2><p>${message}</p></body></html>`,
    { status, headers },
  );
}

function getCookie(req: Request, name: string): string | null {
  const cookie = req.headers.get("cookie") ?? "";
  for (const part of cookie.split(";")) {
    const [key, ...value] = part.trim().split("=");
    if (key === name) return value.join("=");
  }
  return null;
}

Deno.serve(async (req) => {
  if (req.method !== "GET") {
    return page("Method not allowed.", 405);
  }

  const clientId = Deno.env.get("GOOGLE_CLIENT_ID");
  const clientSecret = Deno.env.get("GOOGLE_CLIENT_SECRET");
  const redirectUri = Deno.env.get("GOOGLE_REDIRECT_URI");
  const supabaseUrl = Deno.env.get("SUPABASE_URL");
  const serviceRoleKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");

  if (
    !clientId || !clientSecret || !redirectUri ||
    !supabaseUrl || !serviceRoleKey
  ) {
    console.error("Required OAuth or Supabase environment variables are missing.");
    return page("Server configuration is incomplete.", 500);
  }

  const url = new URL(req.url);
  const code = url.searchParams.get("code");
  const returnedState = url.searchParams.get("state");
  const googleError = url.searchParams.get("error");

  if (googleError) {
    return page("Google authorization was cancelled or denied.", 400, true);
  }

  // Start OAuth when Google has not redirected back with an authorization code.
  if (!code) {
    const stateBytes = crypto.getRandomValues(new Uint8Array(32));
    const state = Array.from(stateBytes, (b) =>
      b.toString(16).padStart(2, "0")
    ).join("");

    const authorize = new URL("https://accounts.google.com/o/oauth2/v2/auth");
    authorize.searchParams.set("client_id", clientId);
    authorize.searchParams.set("redirect_uri", redirectUri);
    authorize.searchParams.set("response_type", "code");
    authorize.searchParams.set(
      "scope",
      "https://www.googleapis.com/auth/gmail.send",
    );
    authorize.searchParams.set("access_type", "offline");
    authorize.searchParams.set("prompt", "consent");
    authorize.searchParams.set("state", state);

    return new Response(null, {
      status: 302,
      headers: {
        "Location": authorize.toString(),
        "Cache-Control": "no-store",
        "Set-Cookie":
          `${COOKIE}=${state}; HttpOnly; Secure; SameSite=Lax; Path=/functions/v1/google-oauth-callback; Max-Age=600`,
      },
    });
  }

  // Validate the OAuth state to protect against forged callbacks.
  const savedState = getCookie(req, COOKIE);
  if (
    !returnedState || !savedState ||
    returnedState.length !== savedState.length
  ) {
    return page("OAuth state validation failed. Start again.", 400, true);
  }

  // Constant-time state comparison.
  const encoder = new TextEncoder();
  const a = encoder.encode(returnedState);
  const b = encoder.encode(savedState);
  let difference = a.length ^ b.length;
  for (let i = 0; i < Math.min(a.length, b.length); i++) {
    difference |= a[i] ^ b[i];
  }
  if (difference !== 0) {
    return page("OAuth state validation failed. Start again.", 400, true);
  }

  try {
    const tokenResponse = await fetch("https://oauth2.googleapis.com/token", {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body: new URLSearchParams({
        code,
        client_id: clientId,
        client_secret: clientSecret,
        redirect_uri: redirectUri,
        grant_type: "authorization_code",
      }),
    });

    if (!tokenResponse.ok) {
      console.error("Google token exchange failed with status", tokenResponse.status);
      return page("Google token exchange failed. Please start again.", 502, true);
    }

    const tokens = await tokenResponse.json();
    if (typeof tokens.refresh_token !== "string" || !tokens.refresh_token) {
      return page(
        "Google did not issue a refresh token. Revoke this app's Google access and repeat consent.",
        400,
        true,
      );
    }

    const supabase = createClient(supabaseUrl, serviceRoleKey, {
      auth: { persistSession: false, autoRefreshToken: false },
    });

    const { error } = await supabase
      .from("google_oauth_credentials")
      .upsert({
        id: "gmail",
        refresh_token: tokens.refresh_token,
        updated_at: new Date().toISOString(),
      });

    if (error) {
      console.error("Could not save OAuth credentials:", error.message);
      return page("The token was obtained but could not be stored securely.", 500, true);
    }

    return page(
      "Google authorization succeeded. The refresh token was saved server-side. You can close this page.",
      200,
      true,
    );
  } catch (error) {
    console.error(
      "OAuth callback failed:",
      error instanceof Error ? error.message : "Unknown error",
    );
    return page("OAuth setup failed. Check the Edge Function logs.", 500, true);
  }
});

import { createClient } from "https://esm.sh/@supabase/supabase-js@2.49.1";

const cors = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, apikey, content-type, x-client-info",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
  "Content-Type": "application/json",
};

function reply(status: number, body: Record<string, unknown>) {
  return new Response(JSON.stringify(body), { status, headers: cors });
}

function base64Url(value: string): string {
  const bytes = new TextEncoder().encode(value);
  let binary = "";
  for (const byte of bytes) binary += String.fromCharCode(byte);
  return btoa(binary).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: cors });
  if (req.method !== "POST") return reply(405, { error: "POST required" });
  if (!req.headers.get("apikey")) return reply(401, { error: "Missing API key" });

  try {
    const body = await req.json();
    const email = typeof body.email === "string" ? body.email.trim().toLowerCase() : "";

    if (email.length > 254 || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      return reply(400, { error: "Enter a valid email address." });
    }

    const url = Deno.env.get("SUPABASE_URL");
    const serviceKey = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");
    const googleClientId = Deno.env.get("GOOGLE_CLIENT_ID");
    const googleClientSecret = Deno.env.get("GOOGLE_CLIENT_SECRET");

    if (!url || !serviceKey || !googleClientId || !googleClientSecret) {
      console.error("Required server secrets are missing.");
      return reply(500, { error: "Email login is not configured." });
    }

    const admin = createClient(url, serviceKey, {
      auth: { autoRefreshToken: false, persistSession: false },
    });

    // Limit each address to one request per minute and five per hour.
    const now = Date.now();
    const { data: limit, error: limitError } = await admin
      .from("email_code_rate_limits")
      .select("last_sent_at, window_started_at, request_count")
      .eq("email", email)
      .maybeSingle();

    if (limitError) throw limitError;

    if (limit) {
      const last = Date.parse(limit.last_sent_at);
      const start = Date.parse(limit.window_started_at);
      if (now - last < 60_000) {
        return reply(429, { error: "Please wait a minute before requesting another code." });
      }
      if (now - start < 3_600_000 && Number(limit.request_count) >= 5) {
        return reply(429, { error: "Too many requests. Try again in an hour." });
      }

      const resetWindow = now - start >= 3_600_000;
      const { error } = await admin.from("email_code_rate_limits").update({
        last_sent_at: new Date(now).toISOString(),
        window_started_at: resetWindow ? new Date(now).toISOString() : limit.window_started_at,
        request_count: resetWindow ? 1 : Number(limit.request_count) + 1,
      }).eq("email", email);
      if (error) throw error;
    } else {
      const { error } = await admin.from("email_code_rate_limits").insert({
        email,
        last_sent_at: new Date(now).toISOString(),
        window_started_at: new Date(now).toISOString(),
        request_count: 1,
      });
      if (error) throw error;
    }

    // Generate a Supabase Auth OTP, not an independent custom password.
    const { data: linkData, error: linkError } = await admin.auth.admin.generateLink({
      type: "magiclink",
      email,
    });

    // Avoid revealing whether an account exists.
    if (linkError || !linkData?.properties?.email_otp) {
      return reply(200, {
        message: "If an eligible account exists, a sign-in code will be sent.",
      });
    }

    const { data: credential, error: credentialError } = await admin
      .from("google_oauth_credentials")
      .select("refresh_token")
      .eq("id", "gmail")
      .single();

    if (credentialError || !credential?.refresh_token) {
      console.error("Gmail refresh token is unavailable.");
      return reply(500, { error: "Email delivery is not configured." });
    }

    const tokenResponse = await fetch("https://oauth2.googleapis.com/token", {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body: new URLSearchParams({
        client_id: googleClientId,
        client_secret: googleClientSecret,
        refresh_token: credential.refresh_token,
        grant_type: "refresh_token",
      }),
    });

    const tokenData = await tokenResponse.json();
    if (!tokenResponse.ok || !tokenData.access_token) {
      console.error("Google access-token refresh failed.");
      return reply(502, { error: "Gmail authorization failed. Reconnect Gmail." });
    }

    const mime = [
      `To: ${email}`,
      "Subject: Your Dinor sign-in code",
      "MIME-Version: 1.0",
      "Content-Type: text/plain; charset=UTF-8",
      "",
      `Your Dinor six-digit sign-in code is: ${linkData.properties.email_otp}`,
      "",
      "Enter this code in the Dinor app to sign in.",
      "If you did not request this code, you can ignore this email.",
    ].join("\r\n");

    const gmailResponse = await fetch(
      "https://gmail.googleapis.com/gmail/v1/users/me/messages/send",
      {
        method: "POST",
        headers: {
          Authorization: `Bearer ${tokenData.access_token}`,
          "Content-Type": "application/json",
        },
        body: JSON.stringify({ raw: base64Url(mime) }),
      },
    );

    if (!gmailResponse.ok) {
      console.error("Gmail API send failed:", gmailResponse.status);
      return reply(502, { error: "Could not send the code. Please try again later." });
    }

    return reply(200, {
      message: "If an eligible account exists, a sign-in code will be sent.",
    });
  } catch (error) {
    console.error("request-email-code failed:", error);
    return reply(500, { error: "Could not request a sign-in code." });
  }
});

import { createClient } from "https://esm.sh/@supabase/supabase-js@2.49.1";

const cors = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Headers": "authorization, apikey, content-type, x-client-info",
  "Access-Control-Allow-Methods": "POST, OPTIONS",
  "Content-Type": "application/json",
};

function reply(status: number, body: Record<string, unknown>) {
  return new Response(JSON.stringify(body), { status, headers: cors });
}

Deno.serve(async (req) => {
  if (req.method === "OPTIONS") return new Response("ok", { headers: cors });
  if (req.method !== "POST") return reply(405, { error: "POST required" });

  // The Flutter client supplies its public Supabase key in the apikey header.
  const apiKey = req.headers.get("apikey");
  if (!apiKey) return reply(401, { error: "Missing API key" });

  try {
    const body = await req.json();
    const email = typeof body.email === "string" ? body.email.trim().toLowerCase() : "";
    const code = typeof body.code === "string" ? body.code.trim() : "";

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email) ||
        !/^\d{8}$/.test(code)) {
      return reply(400, { error: "Enter a valid email and six-digit code." });
    }

    const url = Deno.env.get("SUPABASE_URL");
    if (!url) return reply(500, { error: "Authentication is not configured." });

    // Let Supabase Auth validate its own OTP and issue the real session.
    const response = await fetch(`${url}/auth/v1/verify`, {
      method: "POST",
      headers: {
        apikey: apiKey,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        email,
        token: code,
        type: "magiclink",
      }),
    });

    const result = await response.json();

    if (!response.ok || !result.access_token || !result.refresh_token) {
      return reply(401, { error: "The code is invalid or expired. Request a new code." });
    }

    // These are session credentials intended for the authenticated client.
    return reply(200, {
      access_token: result.access_token,
      refresh_token: result.refresh_token,
      token_type: result.token_type,
      expires_in: result.expires_in,
      user: result.user,
    });
  } catch (error) {
    console.error("verify-email-code failed:", error);
    return reply(500, { error: "Could not verify the code." });
  }
});

package com.dinor.dinor

import android.content.Context
import android.media.Ringtone
import android.media.RingtoneManager
import android.os.Build
import android.os.VibrationEffect
import android.os.Vibrator
import android.os.VibratorManager
import io.flutter.embedding.android.FlutterActivity
import io.flutter.embedding.engine.FlutterEngine
import io.flutter.plugin.common.MethodChannel

class MainActivity : FlutterActivity() {
    private var ringtone: Ringtone? = null

    override fun configureFlutterEngine(flutterEngine: FlutterEngine) {
        super.configureFlutterEngine(flutterEngine)
        MethodChannel(flutterEngine.dartExecutor.binaryMessenger, "dinor/ring")
            .setMethodCallHandler { call, result ->
                when (call.method) {
                    "start" -> { startRinging(); result.success(null) }
                    "stop" -> { stopRinging(); result.success(null) }
                    else -> result.notImplemented()
                }
            }
    }

    private fun vibrator(): Vibrator? =
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.S) {
            (getSystemService(Context.VIBRATOR_MANAGER_SERVICE) as VibratorManager).defaultVibrator
        } else {
            @Suppress("DEPRECATION")
            getSystemService(Context.VIBRATOR_SERVICE) as? Vibrator
        }

    private fun startRinging() {
        stopRinging()
        val uri = RingtoneManager.getDefaultUri(RingtoneManager.TYPE_RINGTONE)
        ringtone = RingtoneManager.getRingtone(applicationContext, uri)?.also { r ->
            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.P) r.isLooping = true
            r.play()
        }
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.O) {
            // Buzz 0.8s, pause 0.6s, repeat until stopped.
            val pattern = longArrayOf(0, 800, 600)
            vibrator()?.vibrate(VibrationEffect.createWaveform(pattern, 0))
        }
    }

    private fun stopRinging() {
        ringtone?.stop()
        ringtone = null
        vibrator()?.cancel()
    }
}
