import 'dart:ffi';
import 'package:ffi/ffi.dart';

typedef _VersionC = Pointer<Utf8> Function();
typedef _VersionD = Pointer<Utf8> Function();

typedef _EncryptC = Int32 Function(Pointer<Utf8>, Pointer<Utf8>, Int32);
typedef _EncryptD = int Function(Pointer<Utf8>, Pointer<Utf8>, int);

typedef _SearchC = Int32 Function(Pointer<Utf8>, Pointer<Utf8>);
typedef _SearchD = int Function(Pointer<Utf8>, Pointer<Utf8>);

typedef _FormatC = Int32 Function(Pointer<Utf8>, Pointer<Utf8>, Int32);
typedef _FormatD = int Function(Pointer<Utf8>, Pointer<Utf8>, int);

typedef _AiExtractC = Int32 Function(Pointer<Utf8>, Pointer<Utf8>, Pointer<Utf8>, Int32);
typedef _AiExtractD = int Function(Pointer<Utf8>, Pointer<Utf8>, Pointer<Utf8>, int);

class DinorCore {
  final DynamicLibrary _lib;
  DinorCore._(this._lib);

  factory DinorCore.load() =>
      DinorCore._(DynamicLibrary.open('libdinor_core.so'));

  String version() {
    final fn = _lib.lookupFunction<_VersionC, _VersionD>('dinor_version');
    return fn().toDartString();
  }

  String encrypt(String text) {
    final fn = _lib.lookupFunction<_EncryptC, _EncryptD>('dinor_encrypt');
    final input = text.toNativeUtf8();
    final out = calloc<Uint8>(text.length + 1).cast<Utf8>();
    final n = fn(input, out, text.length + 1);
    final result = n < 0 ? '' : out.toDartString();
    calloc.free(input);
    calloc.free(out);
    return result;
  }

  // New: Search using C++
  bool search(String haystack, String needle) {
    final fn = _lib.lookupFunction<_SearchC, _SearchD>('dinor_fast_search');
    final h = haystack.toNativeUtf8();
    final n = needle.toNativeUtf8();
    final result = fn(h, n);
    calloc.free(h);
    calloc.free(n);
    return result == 1;
  }

  // New: Format text using C++
  String extractAiAnswer(String query, String webText) {
    final fn = _lib.lookupFunction<_AiExtractC, _AiExtractD>('dinor_extract_web_answer');
    final q = query.toNativeUtf8();
    final w = webText.toNativeUtf8();
    final out = calloc<Uint8>(512).cast<Utf8>(); // Max 512 bytes for answer
    
    final n = fn(q, w, out, 512);
    final result = n < 0 ? '' : out.toDartString();
    
    calloc.free(q);
    calloc.free(w);
    calloc.free(out);
    return result;
  }

  String formatMessage(String text) {
    final fn = _lib.lookupFunction<_FormatC, _FormatD>('dinor_format_message');
    final input = text.toNativeUtf8();
    final out = calloc<Uint8>(text.length + 2).cast<Utf8>();
    final n = fn(input, out, text.length + 2);
    final result = n < 0 ? text : out.toDartString();
    calloc.free(input);
    calloc.free(out);
    return result;
  }
}

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'dart:async';
import 'dart:io';
import 'dart:math';
import '../ffi/dinor_bindings.dart';
import '../chat/ai_service.dart';
import 'package:photo_manager/photo_manager.dart';
import 'ai_avatar.dart';
import 'media_picker.dart';
import 'emoji_picker.dart';
import 'dinor_theme.dart';

class Message {
  final String text;
  final bool isMe;
  final bool isImage;
  final String? caption;
  Message(this.text, this.isMe, {this.isImage = false, this.caption});
}

class ChatScreen extends StatefulWidget {
  final String title;
  final bool isAi;

  const ChatScreen({super.key, required this.title, this.isAi = false});

  @override
  State<ChatScreen> createState() => _ChatScreenState();
}

class _ChatScreenState extends State<ChatScreen> {
  final TextEditingController _textController = TextEditingController();
  final ScrollController _scrollController = ScrollController();
  final List<Message> _messages = [];
  final GlobalKey _inputKey = GlobalKey();
  bool _isAiTyping = false;
  
  // Long press menu state
  Message? _selectedMessage;
  int? _selectedMessageIndex;
  OverlayEntry? _menuOverlay;

  @override
  void initState() {
    super.initState();
    if (widget.isAi) {
      _messages.add(Message("Hello! I am Dinor AI. Ask me anything and I'll search the web.", false));
    }
  }

  void _sendMessage() {
    final rawText = _textController.text.trim();
    if (rawText.isEmpty) return;

    // 1. Format via C++ safely
    String text = rawText;
    try {
      text = DinorCore.load().formatMessage(rawText);
    } catch (e) {
      debugPrint("C++ Format failed, using raw text: $e");
    }

    // 2. Get input position for animation
    final RenderBox? inputBox = _inputKey.currentContext?.findRenderObject() as RenderBox?;
    final Offset startPosition = inputBox != null
        ? inputBox.localToGlobal(Offset.zero)
        : const Offset(0, 0);

    // 3. Clear input field
    _textController.clear();

    // 4. Trigger animation
    _showFlyingMessage(text, startPosition);

    // 5. Add message
    setState(() {
      _messages.add(Message(text, true));
    });
    _scrollToBottom();

    // 6. Trigger AI
    if (widget.isAi) {
      _triggerAiResponse(text);
    }
  }

  void _triggerAiResponse(String userText) async {
    setState(() => _isAiTyping = true);
    _scrollToBottom();

    try {
      String response = await DinorAI.getResponse(userText);
      if (!mounted) return;

      setState(() {
        _isAiTyping = false;
        _messages.add(Message(response, false));
      });
    } catch (e) {
      if (!mounted) return;
      setState(() {
        _isAiTyping = false;
        _messages.add(Message("Error connecting to AI: $e", false));
      });
    }
    _scrollToBottom();
  }

  void _scrollToBottom() {
    Future.delayed(const Duration(milliseconds: 100), () {
      if (_scrollController.hasClients) {
        _scrollController.animateTo(
          _scrollController.position.maxScrollExtent,
          duration: const Duration(milliseconds: 300),
          curve: Curves.easeOut,
        );
      }
    });
  }

  void _showMessageMenu(BuildContext context, int index, Message msg) {
    // Don't open multiple menus at once
    _closeMessageMenu();

    final RenderBox overlay = Overlay.of(context).context.findRenderObject() as RenderBox;
    final RenderBox bubbleBox = context.findRenderObject() as RenderBox;
    final Offset position = bubbleBox.localToGlobal(Offset.zero, ancestor: overlay);

    setState(() {
      _selectedMessage = msg;
      _selectedMessageIndex = index;
    });

    _menuOverlay = OverlayEntry(
      builder: (context) {
        return Stack(
          children: [
            // Invisible full-screen click catcher to dismiss the menu
            Positioned.fill(
              child: GestureDetector(
                onTap: _closeMessageMenu,
                behavior: HitTestBehavior.translucent,
              ),
            ),
            // The Actual Menu
            Positioned(
              top: position.dy - 50,
              left: msg.isMe ? position.dx - 60 : position.dx,
              child: TweenAnimationBuilder<double>(
                tween: Tween(begin: 0.0, end: 1.0),
                duration: const Duration(milliseconds: 150),
                curve: Curves.easeOutBack,
                builder: (context, val, child) {
                  return Transform.scale(
                    scale: val,
                    alignment: msg.isMe ? Alignment.bottomRight : Alignment.bottomLeft,
                    child: child,
                  );
                },
                child: Material(
                  color: Colors.transparent,
                  child: Container(
                    height: 40,
                    padding: const EdgeInsets.symmetric(horizontal: 8),
                    decoration: BoxDecoration(
                      color: Colors.white,
                      borderRadius: BorderRadius.circular(20),
                      boxShadow: [
                        BoxShadow(
                          color: Colors.black.withValues(alpha: 0.15),
                          blurRadius: 10,
                          offset: const Offset(0, 4),
                        )
                      ],
                    ),
                    child: Row(
                      mainAxisSize: MainAxisSize.min,
                      children: [
                        // Forward
                        IconButton(
                          icon: const Icon(Icons.arrow_forward, color: Colors.black54, size: 20),
                          onPressed: () {
                            _closeMessageMenu();
                            ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Forward clicked')));
                          },
                        ),
                        // Copy
                        IconButton(
                          icon: const Icon(Icons.copy, color: Colors.black54, size: 20),
                          onPressed: () {
                            Clipboard.setData(ClipboardData(text: msg.text));
                            _closeMessageMenu();
                            ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Copied to clipboard')));
                          },
                        ),
                        // Delete
                        IconButton(
                          icon: const Icon(Icons.delete, color: Colors.redAccent, size: 20),
                          onPressed: () {
                            setState(() {
                              _messages.removeAt(index);
                            });
                            _closeMessageMenu();
                          },
                        ),
                      ],
                    ),
                  ),
                ),
              ),
            ),
          ],
        );
      },
    );

    Overlay.of(context).insert(_menuOverlay!);
  }

  void _closeMessageMenu() {
    if (_menuOverlay != null) {
      _menuOverlay!.remove();
      _menuOverlay = null;
      setState(() {
        _selectedMessage = null;
        _selectedMessageIndex = null;
      });
    }
  }

  void _showFlyingMessage(String text, Offset startPos) {
    final overlay = Overlay.of(context);
    final overlayEntry = OverlayEntry(
      builder: (context) {
        return _FlyingMessageWidget(
          text: text,
          startPosition: startPos,
          onComplete: () {},
        );
      },
    );
    overlay.insert(overlayEntry);
    Future.delayed(const Duration(milliseconds: 400), () {
      overlayEntry.remove();
    });
  }

  @override
  Widget build(BuildContext context) {
    const Color brandBlue = Color(0xFF229ED9);

    return Scaffold(
      backgroundColor: DinorTheme.bgLight,
      appBar: AppBar(
        backgroundColor: brandBlue,
        foregroundColor: Colors.white,
        leading: _selectedMessage != null
            ? IconButton(
                icon: const Icon(Icons.close),
                onPressed: _closeMessageMenu,
              )
            : null,
        title: Row(
          children: [
            if (widget.isAi && _selectedMessage == null) ...[
              AnimatedAIAvatar(size: 32, isThinking: _isAiTyping),
              const SizedBox(width: 12),
            ],
            if (_selectedMessage != null)
              const Text('Message Options')
            else
              Text(widget.title),
          ],
        ),
      ),
      body: Column(
        children: [
          Expanded(
            child: ListView.builder(
              controller: _scrollController,
              padding: const EdgeInsets.all(16),
              // ignore: deprecated_member_use
              cacheExtent: 1000.0, // Pre-renders messages slightly off-screen for smooth scrolling
              addAutomaticKeepAlives: true, // Keeps images in memory while scrolling
              itemCount: _messages.length + (_isAiTyping ? 1 : 0),
              itemBuilder: (context, index) {
                if (index == _messages.length && _isAiTyping) {
                  return const Align(
                    alignment: Alignment.centerLeft,
                    child: Padding(
                      padding: EdgeInsets.all(8.0),
                      child: Text("Dinor AI is searching...", style: TextStyle(color: Colors.grey, fontStyle: FontStyle.italic)),
                    ),
                  );
                }

                final msg = _messages[index];
                
                Widget messageBubble = Container(
                    margin: const EdgeInsets.only(bottom: 8),
                    padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 10),
                    decoration: BoxDecoration(
                      gradient: msg.isMe ? DinorTheme.primaryGradient : null,
                      color: msg.isMe ? null : DinorTheme.surfaceWhite,
                      borderRadius: BorderRadius.only(
                        topLeft: const Radius.circular(20),
                        topRight: const Radius.circular(20),
                        bottomLeft: msg.isMe ? const Radius.circular(20) : const Radius.circular(4),
                        bottomRight: msg.isMe ? const Radius.circular(4) : const Radius.circular(20),
                      ),
                      boxShadow: msg.isMe ? [
                         BoxShadow(color: brandBlue.withValues(alpha: 0.2), blurRadius: 8, offset: const Offset(0, 4))
                      ] : DinorTheme.softShadow,
                    ),
                    child: msg.isImage
                        ? Column(
                            crossAxisAlignment: CrossAxisAlignment.start,
                            children: [
                              ClipRRect(
                                borderRadius: BorderRadius.circular(8),
                                child: msg.text.startsWith('http')
                                    ? Image.network(
                                        msg.text,
                                        width: 200,
                                        fit: BoxFit.cover,
                                        cacheWidth: 400, // Decodes at half-res to save RAM and GPU
                                        errorBuilder: (context, error, stackTrace) => const Icon(Icons.broken_image, color: Colors.white),
                                      )
                                    : Image.file(
                                        File(msg.text),
                                        width: 200,
                                        fit: BoxFit.cover,
                                        cacheWidth: 400, // Decodes at half-res to save RAM and GPU
                                        errorBuilder: (context, error, stackTrace) => const Icon(Icons.broken_image, color: Colors.white),
                                      ),
                              ),
                              if (msg.caption != null && msg.caption!.isNotEmpty)
                                Padding(
                                  padding: const EdgeInsets.only(top: 8.0),
                                  child: Text(
                                    msg.caption!,
                                    style: TextStyle(
                                      color: msg.isMe ? Colors.white : Colors.black87,
                                      fontSize: 14,
                                    ),
                                  ),
                                ),
                            ],
                          )
                        : Text(
                            msg.text,
                            style: TextStyle(
                              color: msg.isMe ? Colors.white : Colors.black87,
                              fontSize: 16,
                            ),
                          ),
                  );

                // Add the GestureDetector directly to the message bubble
                final interactiveBubble = GestureDetector(
                  onLongPress: () => _showMessageMenu(context, index, msg),
                  child: AnimatedScale(
                    scale: _selectedMessageIndex == index ? 1.1 : 1.0,
                    duration: const Duration(milliseconds: 200),
                    curve: Curves.easeOutBack,
                    child: messageBubble,
                  ),
                );

                // Wrap AI messages with the avatar
                if (!msg.isMe && widget.isAi) {
                  return Row(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      const Padding(
                        padding: EdgeInsets.only(top: 4, right: 8),
                        child: AnimatedAIAvatar(size: 28),
                      ),
                      Flexible(child: interactiveBubble),
                    ],
                  );
                }

                return Align(
                  alignment: msg.isMe ? Alignment.centerRight : Alignment.centerLeft,
                  child: interactiveBubble,
                );
              },
            ),
          ),
          
          Container(
            padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 8),
            color: Colors.white,
            child: SafeArea(
              child: Row(
                crossAxisAlignment: CrossAxisAlignment.end,
                children: [
                  // Media Icon
                  IconButton(
                    icon: const Icon(Icons.add_photo_alternate_outlined, color: Colors.grey),
                    onPressed: () async {
                      final result = await showModalBottomSheet(
                        context: context,
                        isScrollControlled: true,
                        backgroundColor: Colors.transparent,
                        builder: (context) => const CustomMediaPicker(),
                      );

                      if (result != null) {
                        final List<AssetEntity> selectedAssets = result['media'];
                        final String caption = result['caption'];

                        // Add each selected image to the chat
                        for (var asset in selectedAssets) {
                          final file = await asset.file;
                          if (file != null) {
                            setState(() {
                              _messages.add(Message(
                                file.path, 
                                true, 
                                isImage: true, 
                                caption: caption.isNotEmpty ? caption : null
                              ));
                            });
                          }
                        }
                        _scrollToBottom();
                      }
                    },
                  ),
                  
                  // Text Input
                  Expanded(
                    child: Container(
                      key: _inputKey,
                      margin: const EdgeInsets.symmetric(vertical: 6),
                      decoration: BoxDecoration(
                        color: DinorTheme.surfaceWhite,
                        borderRadius: BorderRadius.circular(24),
                        boxShadow: DinorTheme.softShadow,
                      ),
                      child: TextField(
                        controller: _textController,
                        textCapitalization: TextCapitalization.sentences,
                        decoration: const InputDecoration(
                          hintText: 'Message',
                          hintStyle: TextStyle(color: Colors.black45),
                          border: InputBorder.none,
                          contentPadding: EdgeInsets.symmetric(horizontal: 16, vertical: 14),
                        ),
                        onSubmitted: (_) => _sendMessage(),
                      ),
                    ),
                  ),
                  
                  // Emoji & GIF Icon (Merged)
                  IconButton(
                    icon: const Icon(Icons.emoji_emotions_outlined, color: Colors.grey),
                    onPressed: () {
                      showModalBottomSheet(
                        context: context,
                        isScrollControlled: true,
                        backgroundColor: Colors.transparent,
                        builder: (context) => CustomEmojiPicker(
                          onEmojiSelected: (emoji) {
                            // Add the emoji to the text field
                            final newText = _textController.text + emoji;

class _FlyingMessageWidgetState extends State<_FlyingMessageWidget> with SingleTickerProviderStateMixin {

  late Animation<double> _animation;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(vsync: this, duration: const Duration(milliseconds: 300));
    _animation = CurvedAnimation(parent: _controller, curve: Curves.easeInOutCubic);
    _controller.forward().then((_) => widget.onComplete());
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final screenSize = MediaQuery.of(context).size;
    final endPosition = Offset(screenSize.width - 100, screenSize.height * 0.2);
    final controlPoint = Offset(
      widget.startPosition.dx + (endPosition.dx - widget.startPosition.dx) / 2 + 50,
      widget.startPosition.dy - 150,
    );

    return AnimatedBuilder(
      animation: _animation,
      builder: (context, child) {
        final t = _animation.value;
        final x = pow(1 - t, 2) * widget.startPosition.dx + 2 * (1 - t) * t * controlPoint.dx + pow(t, 2) * endPosition.dx;
        final y = pow(1 - t, 2) * widget.startPosition.dy + 2 * (1 - t) * t * controlPoint.dy + pow(t, 2) * endPosition.dy;

        return Positioned(
          left: x.toDouble(),
          top: y.toDouble(),
          child: IgnorePointer(
            child: Opacity(
              opacity: 1.0 - t,
              child: Material(
                color: Colors.transparent,
                child: Container(
                  padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 10),
                  decoration: BoxDecoration(
                    color: const Color(0xFF229ED9),

                  ),
                  child: Text(
                    widget.text,
                    style: const TextStyle(color: Colors.white, fontSize: 16),

            ),

            isThinking: widget.isThinking,
          ),
        );
      },

class _AIAvatarPainter extends CustomPainter {
  final double animationValue;
  final bool isThinking;

  _AIAvatarPainter({required this.animationValue, required this.isThinking});

  @override
  void paint(Canvas canvas, Size size) {
    final center = Offset(size.width / 2, size.height / 2);
    final radius = size.width / 2;
    final t = animationValue;

    // 1. Draw the base 3D Orb (Gradient makes it look spherical)
    final orbPaint = Paint()
      ..shader = const RadialGradient(
        colors: [
          Color(0xFF80E5FF), // Bright highlight
          Color(0xFF24A1DE), // Brand blue
          Color(0xFF005A8A), // Dark shadow edge
        ],
        stops: [0.1, 0.6, 1.0],
        center: Alignment(-0.3, -0.3), // Off-center highlight for 3D effect
        radius: 0.9,
      ).createShader(Rect.fromCircle(center: center, radius: radius));

    canvas.drawCircle(center, radius, orbPaint);

    // 2. Draw rotating 3D rings (Orbits)
    final ringPaint = Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 2.0
      ..color = Colors.white.withValues(alpha: 0.6);

    final ringPaint2 = Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.5
      ..color = const Color(0xFF24A1DE).withValues(alpha: 0.8);

    canvas.save();
    canvas.translate(center.dx, center.dy);

    // Ring 1 (Tilted)
    canvas.save();
    canvas.rotate(t * 2 * pi);
    canvas.drawOval(
      Rect.fromCenter(center: Offset.zero, width: radius * 1.8, height: radius * 0.6),
      ringPaint,
    );
    canvas.restore();

    // Ring 2 (Different angle)
    canvas.save();
    canvas.rotate(-t * 2 * pi + pi / 4);
    canvas.drawOval(
      Rect.fromCenter(center: Offset.zero, width: radius * 1.6, height: radius * 1.2),
      ringPaint2,
    );
    canvas.restore();

    canvas.restore();

    // 3. Draw a glowing "core" in the center
    final corePulse = isThinking ? (sin(t * 10) * 0.2 + 0.8) : 0.5;
    final corePaint = Paint()
      ..color = Colors.white.withValues(alpha: corePulse)
      ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 4);
    
    canvas.drawCircle(center, radius * 0.3, corePaint);

    // 4. Outer glow if thinking
    if (isThinking) {
      final glowPaint = Paint()
        ..color = const Color(0xFF24A1DE).withValues(alpha: 0.3 * corePulse)
        ..style = PaintingStyle.stroke
        ..strokeWidth = 6
        ..maskFilter = const MaskFilter.blur(BlurStyle.normal, 8);
      canvas.drawCircle(center, radius + 4, glowPaint);
    }
  }

  @override
  bool shouldRepaint(covariant _AIAvatarPainter oldDelegate) {
    return oldDelegate.animationValue != animationValue || oldDelegate.isThinking != isThinking;
  }
}

import 'package:connectivity_plus/connectivity_plus.dart';
import 'package:http/http.dart' as http;
import 'package:html/parser.dart' as parser;
import '../ffi/dinor_bindings.dart';

class DinorAI {
  static Future<String> getResponse(String userMessage) async {
    final msg = userMessage.toLowerCase().trim();

    if (msg == 'hi' || msg == 'hello' || msg == 'hey') {
      return "Hello! I am Dinor AI. My brain runs on C++. What do you want to know?";
    }
    if (msg.contains('your name')) {
      return "I am Dinor AI, powered by a native C++ search engine. No external APIs!";
    }

    // Check internet before trying to search
    final connectivityResult = await Connectivity().checkConnectivity();
    if (connectivityResult.contains(ConnectivityResult.none)) {
      return "I am currently offline. Please check your internet connection to search the web!";
    }

    try {
      // 1. Fetch web results
      final query = Uri.encodeComponent(userMessage);
      final url = Uri.parse('https://html.duckduckgo.com/html/?q=$query');
      
      final response = await http.get(url, headers: {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36'
      });

      if (response.statusCode == 200) {
        var document = parser.parse(response.body);
        
        // Get all text from the result snippets
        var snippets = document.getElementsByClassName('result__snippet');
        if (snippets.isEmpty) return "I couldn't find any web results for that.";

        // Combine all text into one giant string
        String allText = snippets.map((e) => e.text.trim()).join(' ');

        // 2. Send to C++ Core to extract the best sentence based on keywords
        final core = DinorCore.load();
        final bestAnswer = core.extractAiAnswer(userMessage, allText);

        if (bestAnswer.isNotEmpty) {
          return "Here is what I found:\n\n$bestAnswer";
        } else {
          // Fallback if C++ couldn't find a good match
          return "I found some results, but they didn't directly answer your question. Here is the raw context:\n\n${allText.substring(0, 150)}...";
        }
      }
      return "My web search failed (Status: ${response.statusCode}).";
    } catch (e) {
      return "Oops, I lost my internet connection. Error: $e";
    }
  }
}

#include "dinor_core.h"
#include <cstring>
#include <cctype>
#include <string>
#include <sstream>
#include <vector>
#include <algorithm>

extern "C" {

const char* dinor_version() {
    return "Dinor 0.3.0 (AI Core)";
}

int dinor_encrypt(const char* plaintext, char* out, int out_len) {
    int n = strlen(plaintext);
    if (n + 1 > out_len) return -1;
    for (int i = 0; i < n; i++) out[i] = plaintext[n - 1 - i];
    out[n] = '\0';
    return n;
}

int dinor_fast_search(const char* haystack, const char* needle) {
    if (!*needle) return 1;
    for (const char* h = haystack; *h; ++h) {
        const char* h_ptr = h;
        const char* n_ptr = needle;
        while (*h_ptr && *n_ptr && (tolower(*h_ptr) == tolower(*n_ptr))) {
            h_ptr++; n_ptr++;
        }
        if (!*n_ptr) return 1;
    }
    return 0;
}

int dinor_format_message(const char* input, char* out, int out_len) {
    std::string result;
    bool last_was_space = true;
    for (const char* p = input; *p; ++p) {
        if (*p == ' ') {
            if (!last_was_space) { result += ' '; last_was_space = true; }
        } else {
            result += *p; last_was_space = false;
        }
    }
    if (!result.empty() && result.back() == ' ') result.pop_back();
    if (result.length() + 1 > out_len) return -1;
    strcpy(out, result.c_str());
    return result.length();
}

// AI Logic: Find the sentence in the web text that best matches the user's query
int dinor_extract_web_answer(const char* query, const char* html_text, char* out, int out_len) {
    std::string text(html_text);
    std::string q(query);
    std::string best_sentence = "";
    int max_matches = 0;

    // Split HTML/text into sentences (naive splitting by punctuation)
    std::stringstream ss(text);
    std::string sentence;
    
    // Clean the query into words
    std::vector<std::string> keywords;
    std::stringstream qss(q);
    std::string word;
    while (qss >> word) {
        // Remove punctuation from keyword
        word.erase(std::remove_if(word.begin(), word.end(), ::ispunct), word.end());
        if (word.length() > 2) keywords.push_back(word);
    }

    while (std::getline(ss, sentence, '.')) {
        // Trim leading spaces
        sentence.erase(0, sentence.find_first_not_of(" \n\r\t"));
        sentence.erase(sentence.find_last_not_of(" \n\r\t") + 1);

        if (sentence.length() < 20 || sentence.length() > 300) continue;

        int matches = 0;
        for (const auto& kw : keywords) {
            if (dinor_fast_search(sentence.c_str(), kw.c_str())) matches++;
        }

        if (matches > max_matches) {
            max_matches = matches;
            best_sentence = sentence + ".";
        }
    }

    if (best_sentence.empty()) return -1; // No good sentence found

    if (best_sentence.length() + 1 > out_len) return -1;
    strcpy(out, best_sentence.c_str());
    return best_sentence.length();
}

}
